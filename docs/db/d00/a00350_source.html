77static bool valid_access(uint32_t a, unsigned bits, uint32_t limit)
78{
79 return (bits == 8 || bits == 16 || bits == 32) && (a % (bits / 8) == 0) &&

107static bool source_rejected(
108 uint32_t control, uint32_t key, uint32_t mask, uint32_t source,
109 uint32_t bytes)
110{
111 if (!control)
112 return false;
113 if (bytes < 4)
114 mask &= (1u << (8 * bytes)) - 1;
115 const bool equal = ((source ^ key) & mask) == 0;
116 return (control & 7) == 4 ? equal : !equal;
117}

119// Select the bit order within each byte; byte order itself is unchanged.
120static bool mono_bit(uint32_t word, unsigned pixel, bool lsb)
121{
122 const unsigned bit =
123 (pixel & ~7u) + (lsb ? (pixel & 7u) : 7u - (pixel & 7u));
124 return ((word >> bit) & 1u) != 0;
125}

137uint32_t CRV200::ReadMem(uint32_t a, int dsize)
138{
139 m_counters.reads++;
140 const unsigned bits = unsigned(dsize);
141 if (!valid_access(a, bits, RegisterSpaceSize))
142 {
143 fail("invalid MMIO read " + hex(a));
144 return 0xffffffffu;
145 }
146 if (a >= 0x3b0 && a < 0x3e0)
147 {
148 if (!m_vga_read)
149 {
150 fail("VGA register frontend is not connected");
151 return field_mask(bits);
152 }
153 uint32_t value = 0;
154 for (unsigned i = 0; i < bits / 8; ++i)
155 value |= uint32_t(m_vga_read(a + i)) << (8 * i);
156 return value;
157 }
158 if ((a & ~3u) == MM_DATA)
159 {
160 const uint32_t idx = r(MM_INDEX);
161 uint32_t target;
162 if (idx & 0x80000000u)
163 return uint32_t(mem_read((idx & 0x7ffffffcu) + (a & 3u), dsize));
164 return indexed_target(a & 3u, target) ? ReadMem(target, dsize) : 0;
165 }
166 // Side-effect ports cannot be split implicitly into multiple reads.
167 const uint32_t base = alias(a & ~3u), shift = (a & 3u) * 8;
168 if (base == PALETTE_DATA && bits != 32)
169 {
170 fail("palette data requires DWORD access");
171 return 0;
172 }
173 return (read32(base) >> shift) & field_mask(bits);
174}

176void CRV200::WriteMem(uint32_t a, int dsize, uint32_t value)
177{
178 m_counters.writes++;
179 const unsigned bits = unsigned(dsize);
180 if (!valid_access(a, bits, RegisterSpaceSize))
181 {
182 fail("invalid MMIO write " + hex(a));
183 return;
184 }
185 if (a >= 0x3b0 && a < 0x3e0)
186 {
187 if (!m_vga_write)
188 {
189 fail("VGA register frontend is not connected");
190 return;
191 }
192 for (unsigned i = 0; i < bits / 8; ++i)
193 m_vga_write(a + i, uint8_t(value >> (8 * i)));
194 return;
195 }
196 if ((a & ~3u) == MM_DATA)
197 {
198 const uint32_t idx = r(MM_INDEX);
199 uint32_t target;
200 if (idx & 0x80000000u)
201 mem_write((idx & 0x7ffffffcu) + (a & 3u), dsize, value);
202 else if (indexed_target(a & 3u, target))
203 WriteMem(target, dsize, value);
204 return;
205 }
206 const uint32_t base = alias(a & ~3u), shift = (a & 3u) * 8,
207 mask = field_mask(bits) << shift;
208 if ((base == PALETTE_DATA || base == DST_HEIGHT_WIDTH ||
209 base == DST_WIDTH_HEIGHT || base == DST_LINE_END ||
210 (base >= HOST_DATA0 && base <= HOST_DATA_LAST)) &&
211 bits != 32)
212 {
213 fail("command/data port requires DWORD access at " + hex(a));
214 return;
215 }
216 write32(base, (value << shift) & mask, mask);
217}

219// MM_DATA is a window onto the register MM_INDEX selects; its byte lanes
220// address the matching bytes of that register, as the x86 BIOS uses them.
221bool CRV200::indexed_target(uint32_t lane, uint32_t& target)
222{
223 target = (r(MM_INDEX) & ~3u) + lane;
224 if (target >= RegisterSpaceSize)
225 {
226 fail("MM_DATA register index outside MMIO aperture");
227 return false;
228 }
229 if (target < 8)
230 {
231 fail("recursive MM_DATA index");
232 return false;
233 }
234 return true;
235}

237uint32_t CRV200::read32(uint32_t a)
238{
239 if (a == CLOCK_CNTL_DATA)
240 {
241 const uint32_t i = r(CLOCK_CNTL_INDEX) & 63, v = m_pll[i];
242 // PLL_TEST_CNTL[31:24] counts 27 MHz reference clocks; the x86 BIOS
243 // times its delays with it. Each read observes one more microsecond.
244 if (i == PLL_TEST_CNTL)
245 m_pll[i] += 27u << 24;
246 return v;
247 }
248 if (a == PALETTE_DATA)
249 {
250 const uint32_t idx = (r(PALETTE_INDEX) >> 16) & 255;
251 const uint32_t val = m_palette[idx];
252 r(PALETTE_INDEX) =
253 (r(PALETTE_INDEX) & ~0xff0000u) | (((idx + 1) & 255) << 16);
254 return val;
255 }
256 if (a == SRC_PITCH || a == DST_PITCH)
257 return (r(a == SRC_PITCH ? SRC_PITCH_OFFSET : DST_PITCH_OFFSET) >> 16) &
258 0x3fc0u;
259 if (a == SRC_OFFSET || a == DST_OFFSET)
260 return (r(a == SRC_OFFSET ? SRC_PITCH_OFFSET : DST_PITCH_OFFSET) &
261 0x3fffffu)
262 << 10;
263 if (a == SRC_X || a == DST_X)
264 return r(a == SRC_X ? SRC_Y_X : DST_Y_X) & 0xffffu;
265 if (a == SRC_Y || a == DST_Y)
266 return r(a == SRC_Y ? SRC_Y_X : DST_Y_X) >> 16;
267 if (a == DP_CNTL_XDIR_YDIR_YMAJOR)
268 return ((r(DP_CNTL) & 1u) << 31) | ((r(DP_CNTL) & 2u) << 14) |
269 (r(DP_CNTL) & 4u);
270 // Synchronous PIO availability, NOT a cycle model.
271 if (a == RBBM_STATUS)
272 return (busy() ? GUI_ACTIVE : 0u) | (halted() ? 0u : 64u);
273 // Both channels powered up (bits 1:0, polled by the BIOS) and idle; no
274 // DRAM timing.
275 if (a == MC_STATUS)
276 return 7u;
277 if (a == CRTC_STATUS)
278 {
279 // VBLANK_SAVE latches at vertical blank start until written with 1.
280 const bool vblank =
281 m_scanline >= (((r(CRTC_V_TOTAL_DISP) >> 16) & 0xfff) + 1);
282 return (vblank ? CRTC_VBLANK_CUR : 0u) |
283 (r(CRTC_STATUS) & CRTC_VBLANK_SAVE);
284 }
285 if (a == CRTC_VLINE_CRNT_VLINE)
286 return (m_scanline & 0x1fff) << 16;
287 if (a == CRTC_CRNT_FRAME)
288 return m_frame;
289 if (a == DSTCACHE_CTLSTAT)
290 return busy() ? 0x80000000u : 0;
291 // The CP never consumes commands in this model.
292 if (a == CP_RB_RPTR)
293 return 0;
294 if (a >= 0xf00 && a < 0x1000)
295 return m_config_read ? m_config_read(a - 0xf00) : 0xffffffffu;
296 if (a >= GPIO_VGA_DDC && a <= GPIO_CRT2_DDC)
297 {
298 // Disconnected DDC: open-drain pull-ups, no EDID responder or ACK.
299 uint32_t v = r(a) & ~0x300u;
300 if (!(v & (1u << 16)) || (v & 1u))
301 v |= 1u << 8;
302 if (!(v & (1u << 17)) || (v & 2u))
303 v |= 1u << 9;
304 return v;
305 }
306 return r(a);
307}

309void CRV200::write32(uint32_t a, uint32_t val, uint32_t lanes)
310{
311 const uint32_t old = r(a), v = (old & ~lanes) | (val & lanes);
312 if (a >= 0xf00 && a < 0x1000)
313 {
314 note("read-only PCI configuration mirror write " + hex(a));
315 return;
316 }
317 if (a == CLOCK_CNTL_INDEX)
318 {
319 // Keep PLL_DIV_SEL [9:8], not only the index/write flag.
320 r(a) = v & 0x3bfu;
321 return;
322 }
323 if (a == CLOCK_CNTL_DATA)
324 {
325 // Byte lanes update only their part of the selected PLL register.
326 uint32_t& pll = m_pll[r(CLOCK_CNTL_INDEX) & 63];
327 if (r(CLOCK_CNTL_INDEX) & 0x80)
328 pll = (pll & ~lanes) | (val & lanes);
329 return;
330 }
331 if (a == CRTC_STATUS)
332 {
333 r(a) &= ~(val & lanes & CRTC_VBLANK_SAVE);
334 return;
335 }
336 if (a == DAC_CNTL)
337 {
338 // The board's VGA frontend consumes DAC_8BIT_EN; other fields are stored.
339 r(a) = v;
340 return;
341 }
342 if (a == PALETTE_INDEX)
343 {
344 r(a) = v & 0x00ff00ffu;
345 if (m_palette_index)
346 m_palette_index();
347 return;
348 }
349 if (a == PALETTE_DATA)
350 {
351 const uint32_t idx = r(PALETTE_INDEX) & 255;
352 m_palette[idx] = v & 0xffffff;
353 r(PALETTE_INDEX) = (r(PALETTE_INDEX) & ~255u) | ((idx + 1) & 255);
354 return;
355 }
356 if (a == GEN_INT_STATUS)
357 {
358 r(a) &= ~(val & lanes & 1u);
359 update_irq();
360 return;
361 }
362 if (a == GEN_INT_CNTL)
363 {
364 r(a) = v;
365 update_irq();
366 return;
367 }
368 if (a == RBBM_SOFT_RESET)
369 {
370 if ((v & ~old) & (1u | 32u))
371 {
372 m_pending = Pending{};
373 m_fault.clear();
374 }
375 r(a) = v;
376 return;
377 }
378 if (a == RBBM_STATUS || a == MC_STATUS || a == CONFIG_MEMSIZE ||
379 a == CONFIG_APER_SIZE || a == CONFIG_APER_0_BASE ||
380 a == CONFIG_APER_1_BASE || a == CONFIG_REG_1_BASE ||
381 a == CONFIG_REG_APER_SIZE || a == CRTC_CRNT_FRAME || a == CP_RB_RPTR)
382 return;
383 if (a == CP_RB_WPTR || a == CP_IB_BUFSZ || a == CP_CSQ_CNTL)
384 {
385 r(a) = v;
386 if (v)
387 fail(
388 "CP/ring/indirect-buffer execution is unsupported at " +
389 hex(a));
390 return;
391 }
392 if (a >= 0x1000 && a < 0x1400)
393 {
394 r(a) = v;
395 if (v)
396 fail("CP command FIFO unsupported");
397 return;
398 }
399 if (a == DST_LINE_END)
400 {
401 r(a) = v;
402 if (!halted())
403 draw_line(r(DST_LINE_START), v);
404 return;
405 }
406 if (a == RB3D_CNTL)
407 {
408 r(a) = v;
409 if (v)
410 fail("3D pipeline unsupported");
411 return;
412 }
413 if (a == CRTC2_GEN_CNTL)
414 {
415 r(a) = v;
416 if (v & CRTC_EN)
417 fail("second CRTC unsupported");
418 return;
419 }
420 if (a == WAIT_UNTIL)
421 {
422 r(a) = v;
423 if (v & ~(WAIT_2D | WAIT_DMA))
424 note("WAIT_UNTIL contains unmodeled conditions " + hex(v));
425 return;
426 }
427 // Immediate flush of software-backed memory.
428 if (a == DSTCACHE_CTLSTAT)
429 {
430 r(a) = v & ~5u;
431 return;
432 }
433 if (a == MM_INDEX || a == MC_FB_LOCATION || a == SURFACE_CNTL ||
434 a == BUS_CNTL || a == HOST_PATH_CNTL || a == CRTC_H_TOTAL_DISP ||
435 a == CRTC_V_TOTAL_DISP || a == CRTC_GEN_CNTL || a == CRTC_EXT_CNTL ||
436 a == CRTC_OFFSET || a == CRTC_OFFSET_CNTL || a == CRTC_PITCH ||
437 (a >= CUR_OFFSET && a <= CUR_CLR1) || a == DP_WRITE_MSK ||
438 a == DP_CNTL || a == DEFAULT_PITCH_OFFSET ||
439 a == DEFAULT_SC_BOTTOM_RIGHT || a == SC_TOP_LEFT ||
440 a == SC_BOTTOM_RIGHT || a == SRC_SC_BOTTOM_RIGHT || a == DP_DATATYPE ||
441 a == CLR_CMP_CLR_SRC || a == CLR_CMP_CLR_DST || a == CLR_CMP_MASK ||
442 a == BRUSH_Y_X || a == BRUSH_DATA0 || a == BRUSH_DATA1 ||
443 a == DST_LINE_START || a == DST_LINE_PATCOUNT || a == DP_MIX ||
444 a == DP_BRUSH_FRGD_CLR || a == DP_BRUSH_BKGD_CLR ||
445 a == DP_SRC_FRGD_CLR || a == DP_SRC_BKGD_CLR || a == CLR_CMP_CNTL ||
446 a == DP_SRC_ENDIAN || (a >= GPIO_VGA_DDC && a <= GPIO_CRT2_DDC))
447 {
448 r(a) = v;
449 return;
450 }
451 if (a == DP_GUI_MASTER_CNTL)
452 {
453 r(a) = v;
454 r(DP_DATATYPE) =
455 ((v >> 8) & 15) | ((v & 0x30f0u) << 4) | ((v & 0x4000u) << 16);
456 r(DP_MIX) = (v & 0xff0000u) | ((v >> 16) & 0x700u);
457 if (!(v & GMC_SRC_PITCH))
458 r(SRC_PITCH_OFFSET) = r(DEFAULT_PITCH_OFFSET);
459 if (!(v & GMC_DST_PITCH))
460 r(DST_PITCH_OFFSET) = r(DEFAULT_PITCH_OFFSET);
461 if (!(v & GMC_DST_CLIP))
462 {
463 r(SC_TOP_LEFT) = 0;
464 r(SC_BOTTOM_RIGHT) = r(DEFAULT_SC_BOTTOM_RIGHT);
465 }
466 if (!(v & GMC_SRC_CLIP))
467 r(SRC_SC_BOTTOM_RIGHT) = r(DEFAULT_SC_BOTTOM_RIGHT);
468 // GUI_MASTER is a setup alias: disable resets the compare control.
469 // A subsequent explicit CLR_CMP_CNTL write can enable source keying.
470 if (v & GMC_CLR_CMP_DISABLE)
471 r(CLR_CMP_CNTL) = 0;
472 if (v & GMC_WR_MSK_DISABLE)
473 r(DP_WRITE_MSK) = 0xffffffffu;
474 return;
475 }
476 if (a == DP_CNTL_XDIR_YDIR_YMAJOR)
477 {
478 r(a) = v;
479 r(DP_CNTL) =
480 (r(DP_CNTL) & ~7u) | ((v >> 31) & 1u) | ((v >> 14) & 2u) | (v & 4u);
481 return;
482 }
483 if (a == SRC_PITCH_OFFSET || a == DST_PITCH_OFFSET)
484 {
485 r(a) = v;
486 return;
487 }
488 if (a == SRC_OFFSET || a == DST_OFFSET)
489 {
490 const uint32_t t =
491 a == SRC_OFFSET ? SRC_PITCH_OFFSET : DST_PITCH_OFFSET;
492 if (v & 1023u)
493 {
494 fail("sub-kilobyte engine offset unsupported");
495 return;
496 }
497 r(t) = (r(t) & 0xffc00000u) | ((v >> 10) & 0x3fffffu);
498 r(a) = v;
499 return;
500 }
501 if (a == SRC_PITCH || a == DST_PITCH)
502 {
503 if (v & 63u)
504 {
505 fail("non-64-byte source pitch unsupported");
506 return;
507 }
508 const uint32_t target =
509 a == SRC_PITCH ? SRC_PITCH_OFFSET : DST_PITCH_OFFSET;
510 r(target) = (r(target) & 0xc03fffffu) | ((v << 16) & 0x3fc00000u);
511 r(a) = v;
512 return;
513 }
514 if (a == SRC_X || a == SRC_Y || a == DST_X || a == DST_Y)
515 {
516 r(a) = v;
517 const uint32_t target = (a == SRC_X || a == SRC_Y) ? SRC_Y_X : DST_Y_X;
518 r(target) = (a == SRC_X || a == DST_X)
519 ? ((r(target) & 0xffff0000u) | (v & 0xffff))
520 : ((r(target) & 0xffffu) | (v << 16));
521 return;
522 }
523 if (a == SRC_Y_X || a == DST_Y_X)
524 {
525 r(a) = v;
526 return;
527 }
528 if (a == SRC_X_Y || a == DST_X_Y)
529 {
530 r(a) = v;
531 r(a == SRC_X_Y ? SRC_Y_X : DST_Y_X) = (v << 16) | (v >> 16);
532 return;
533 }
534 if (a == SC_LEFT || a == SC_TOP)
535 {
536 r(a) = v;
537 r(SC_TOP_LEFT) = a == SC_LEFT
538 ? ((r(SC_TOP_LEFT) & 0xffff0000u) | (v & 0xffff))
539 : ((r(SC_TOP_LEFT) & 0xffff) | (v << 16));
540 return;
541 }
542 if (a == SC_RIGHT || a == SC_BOTTOM)
543 {
544 r(a) = v;
545 r(SC_BOTTOM_RIGHT) = a == SC_RIGHT
546 ? ((r(SC_BOTTOM_RIGHT) & 0xffff0000u) | (v & 0xffff))
547 : ((r(SC_BOTTOM_RIGHT) & 0xffff) | (v << 16));
548 return;
549 }
550 if (a == DST_HEIGHT_WIDTH || a == DST_WIDTH_HEIGHT)
551 {
552 r(a) = v;
553 if (halted())
554 return;
555 draw(
556 a == DST_HEIGHT_WIDTH ? v & 0xffff : v >> 16,
557 a == DST_HEIGHT_WIDTH ? v >> 16 : v & 0xffff);
558 return;
559 }
560 if (a >= HOST_DATA0 && a <= HOST_DATA_LAST)
561 {
562 r(a) = v;
563 host_word(v, a == HOST_DATA_LAST);
564 return;
565 }
566 if (!shadow_allowed(a) && m_options.strict_mmio)
567 {
568 fail("unknown MMIO write " + hex(a));
569 return;
570 }
571 r(a) = v;
572 m_counters.shadow_writes++;
573 // Stored without a modeled effect: report the register once.
574 if (!m_warned[a / 4])
575 {
576 m_warned[a / 4] = true;
577 if (m_unimplemented)
578 m_unimplemented("RV200 register", a, v, true);
579 }
580}

584// Aperture 1 mirrors aperture 0 with its own NONSURF byte swapping, unless
585// HOST_PATH_CNTL.HDP_APER_CNTL maps it to the following VRAM instead.
586bool CRV200::aperture_index(uint64_t a, size_t& index)
587{
588 if (a < 2ull * ApertureSize)
589 {
590 const bool ap1 = a >= ApertureSize;
591 const uint32_t flags = (r(SURFACE_CNTL) >> (ap1 ? 22 : 20)) & 3u;
592 const uint32_t sw = (flags & 1u) ^ ((flags & 2u) ? 3u : 0u);
593 uint64_t offset = a;
594 if (ap1 && !(r(HOST_PATH_CNTL) & HDP_APER_CNTL))
595 offset -= ApertureSize;
596 offset ^= sw;
597 if (offset < m_vram.size())
598 {
599 index = size_t(offset);
600 return true;
601 }
602 }
603 if (!m_aperture_warned)
604 {
605 m_aperture_warned = true;
606 note("aperture access beyond implemented VRAM " + hex(uint32_t(a)));
607 }
608 return false;
609}

611uint64_t CRV200::mem_read(uint32_t a, int dsize)
612{
613 if (dsize != 8 && dsize != 16 && dsize != 32 && dsize != 64)
614 {
615 fail("invalid aperture read size");
616 return ~uint64_t(0);
617 }
618 // Unimplemented VRAM reads as open bus.
619 uint64_t v = 0;
620 for (unsigned i = 0; i < unsigned(dsize) / 8; i++)
621 {
622 size_t index = 0;
623 const uint8_t byte =
624 aperture_index(uint64_t(a) + i, index) ? m_vram[index] : 0xff;
625 v |= uint64_t(byte) << (8 * i);
626 }
627 return v;
628}

630void CRV200::mem_write(uint32_t a, int dsize, uint64_t value)
631{
632 if (dsize != 8 && dsize != 16 && dsize != 32 && dsize != 64)
633 {
634 fail("invalid aperture write size");
635 return;
636 }
637 for (unsigned i = 0; i < unsigned(dsize) / 8; i++)
638 {
639 size_t index = 0;
640 if (aperture_index(uint64_t(a) + i, index))
641 m_vram[index] = uint8_t(value >> (8 * i));
642 }
643}

645bool CRV200::translate(uint64_t gpu, size_t count, size_t& index) const
646{
647 const uint64_t base = uint64_t(r(MC_FB_LOCATION) & 0xffff) << 16;
648 const uint64_t end = (uint64_t(r(MC_FB_LOCATION) >> 16) + 1) << 16;
649 if (gpu < base || gpu > end || count > end - gpu ||
650 gpu - base > m_vram.size() || count > m_vram.size() - (gpu - base))
651 return false;
652 index = size_t(gpu - base);
653 return true;
654}

656bool CRV200::pixel_address(
657 uint32_t base, uint32_t pitch, int32_t x, int32_t y, uint32_t bpp,
658 size_t& idx) const
659{
660 if (x < 0 || y < 0 || uint64_t(x) * bpp + bpp > pitch)
661 return false;
662 return translate(
663 uint64_t(base) + uint64_t(y) * pitch + uint64_t(x) * bpp, bpp, idx);
664}

666uint32_t CRV200::pixel_read(size_t i, uint32_t b) const
667{
668 uint32_t v = 0;
669 for (uint32_t j = 0; j < b; j++)
670 v |= uint32_t(m_vram[i + j]) << (8 * j);
671 return v;
672}

674void CRV200::pixel_write(size_t i, uint32_t b, uint32_t v)
675{
676 for (uint32_t j = 0; j < b; j++)
677 m_vram[i + j] = uint8_t(v >> (8 * j));
678 m_counters.pixels++;
679}

681uint32_t CRV200::rop3(uint8_t op, uint32_t p, uint32_t s, uint32_t d)
682{
683 uint32_t out = 0;
684 for (unsigned i = 0; i < 8; i++)
685 if (op & (1u << i))
686 out |= (i & 4 ? p : ~p) & (i & 2 ? s : ~s) & (i & 1 ? d : ~d);
687 return out;
688}

690// Scissor edges are 14-bit sign-magnitude; right/bottom are exclusive (XAA
691// writes x2 + 1).
692static int32_t scissor_edge(uint32_t v)
693{
694 return (v & 0x8000) ? -int32_t(v & 0x3fff) : int32_t(v & 0x3fff);
695}
696
697bool CRV200::clipped(int32_t x, int32_t y, uint32_t tl, uint32_t br) const
698{
699 return x < scissor_edge(tl) || y < scissor_edge(tl >> 16) ||
700 x >= scissor_edge(br) || y >= scissor_edge(br >> 16);
701}

703void CRV200::draw(uint32_t w, uint32_t h)
704{
705 if (m_pending.active)
706 {
707 fail("new primitive while host upload is pending");
708 return;
709 }
710 if (!w || !h)
711 return;
712 if (w > 8192 || h > 8192 || uint64_t(w) * h > 16u * 1024 * 1024)
713 {
714 fail("primitive exceeds bounded work limit");
715 return;
716 }
717 const uint32_t g = r(DP_GUI_MASTER_CNTL), type = r(DP_DATATYPE),
718 bpp = bytes_per_pixel(type & 15);
719 const uint8_t op = uint8_t(r(DP_MIX) >> 16);
720 bool uses_s = rop_uses(op, 1);
721 const bool uses_p = rop_uses(op, 2);
722 const uint32_t source = (r(DP_MIX) >> 8) & 7, brush = (type >> 8) & 15;
723 if (!bpp)
724 {
725 fail("unsupported pixel format");
726 return;
727 }
728 if ((g & (1u << 27)) || !compare_supported(r(CLR_CMP_CNTL)))
729 {
730 fail("3D or unsupported color comparison in 2D operation");
731 return;
732 }
733 const uint32_t compare = r(CLR_CMP_CNTL);
734 const uint32_t key = r(CLR_CMP_CLR_SRC), key_mask = r(CLR_CMP_MASK);
735 // A compare can require source data even when the selected ROP does not;
736 // host input is consumed even for a source-independent ROP.
737 if (compare || source == 3)
738 uses_s = true;
739 if (g & GMC_SRC_CLIP)
740 {
741 fail("source clipping unsupported");
742 return;
743 }
744 if (uses_p && brush != 13 && brush != 0 && brush != 1)
745 {
746 fail("unsupported pattern brush datatype");
747 return;
748 }
749 if (uses_p && brush <= 1 && (g & 0x80000000u))
750 {
751 fail("8x8 brush origin auto-load not implemented");
752 return;
753 }
754 if (uses_s && source != 2 && source != 3)
755 {
756 fail("unsupported 2D source selector");
757 return;
758 }
759 const uint32_t source_format = (type >> 16) & 3;
760 const bool mono = source_format < 2;
761 if (uses_s && source_format != 3 && !(source == 3 && mono))
762 {
763 fail("unsupported source datatype/transport");
764 return;
765 }
766 if (uses_s && source == 3 && uses_p && brush != 13)
767 {
768 fail("combined host transfer and patterned brush unsupported");
769 return;
770 }
771 if ((type & 0x20000000u) || r(DP_SRC_ENDIAN))
772 {
773 fail("host/source endianness mode unsupported");
774 return;
775 }
776 const uint32_t dst = r(DST_PITCH_OFFSET), src = r(SRC_PITCH_OFFSET);
777 if ((dst & 0xc0000000u) || (uses_s && source == 2 && (src & 0xc0000000u)))
778 {
779 fail("tiled engine surfaces unsupported");
780 return;
781 }
782 const uint32_t db = (dst & 0x3fffffu) << 10, dp = (dst >> 16) & 0x3fc0u;
783 const uint32_t sb = (src & 0x3fffffu) << 10, sp = (src >> 16) & 0x3fc0u;
784 const int32_t x = int16_t(r(DST_Y_X) & 0xffff),
785 y = int16_t(r(DST_Y_X) >> 16);
786 const int32_t sx = int16_t(r(SRC_Y_X) & 0xffff),
787 sy = int16_t(r(SRC_Y_X) >> 16);
788 const int32_t xd = (r(DP_CNTL) & 1) ? 1 : -1,
789 yd = (r(DP_CNTL) & 2) ? 1 : -1;
790 const uint32_t tl = r(SC_TOP_LEFT), br = r(SC_BOTTOM_RIGHT),
791 mask = r(DP_WRITE_MSK), p = r(DP_BRUSH_FRGD_CLR);
792 const uint32_t brush_x = r(BRUSH_Y_X) & 7,
793 brush_y = (r(BRUSH_Y_X) >> 8) & 7;
794 // Validate the entire destination/source footprint before modifying VRAM.
795 for (uint32_t j = 0; j < h; j++)
796 for (uint32_t i = 0; i < w; i++)
797 {
798 const int32_t xx = x + xd * int32_t(i), yy = y + yd * int32_t(j);
799 size_t di = 0, si = 0;
800 if (clipped(xx, yy, tl, br))
801 continue;
802 if (!pixel_address(db, dp, xx, yy, bpp, di) ||
803 (uses_s && source == 2 &&
804 !pixel_address(
805 sb, sp, sx + xd * int32_t(i), sy + yd * int32_t(j), bpp,
806 si)))
807 {
808 fail("2D address outside GPU framebuffer window/pitch");
809 return;
810 }
811 }
812 m_counters.draws++;
813 if (uses_s && source == 3)
814 {
815 if (bpp == 3 && !mono)
816 {
817 fail("24-bit HOST_DATA packing unsupported");
818 return;
819 }
820 Pending& q = m_pending;
821 q = Pending{};
822 q.active = true;
823 q.width = w;
824 q.height = h;
825 q.words_per_row = mono ? (w + 31) / 32 : (w * bpp + 3) / 4;
826 q.x = x;
827 q.y = y;
828 q.dx = xd;
829 q.dy = yd;
830 q.base = db;
831 q.pitch = dp;
832 q.bpp = bpp;
833 q.mask = mask;
834 q.pattern = p;
835 q.rop = op;
836 q.sc_tl = tl;
837 q.sc_br = br;
838 q.source_format = source_format;
839 q.mono_lsb = (type >> 30) & 1u;
840 q.fg = r(DP_SRC_FRGD_CLR);
841 q.bg = r(DP_SRC_BKGD_CLR);
842 q.compare = compare;
843 q.compare_color = key;
844 q.compare_mask = key_mask;
845 return;
846 }
847 for (uint32_t j = 0; j < h; j++)
848 for (uint32_t i = 0; i < w; i++)
849 {
850 const int32_t xx = x + xd * int32_t(i), yy = y + yd * int32_t(j);
851 size_t di = 0, si = 0;
852 if (clipped(xx, yy, tl, br))
853 continue;
854 pixel_address(db, dp, xx, yy, bpp, di);
855 uint32_t s = 0;
856 if (uses_s && source == 2)
857 {
858 pixel_address(
859 sb, sp, sx + xd * int32_t(i), sy + yd * int32_t(j), bpp,
860 si);
861 s = pixel_read(si, bpp);
862 }
863 if (source_rejected(compare, key, key_mask, s, bpp))
864 continue;
865 uint32_t pattern = p;
866 if (uses_p && brush <= 1)
867 {
868 // Screen-aligned pattern; BRUSH_Y_X offsets its origin.
869 const unsigned row = unsigned(yy + int32_t(brush_y)) & 7u;
870 const unsigned col = unsigned(xx + int32_t(brush_x)) & 7u;
871 const uint32_t data = r(row < 4 ? BRUSH_DATA0 : BRUSH_DATA1);
872 const bool ink = mono_bit(
873 data, (row & 3u) * 8u + col, (type & (1u << 30)) != 0);
874 if (!ink && brush == 1)
875 continue;
876 pattern = ink ? p : r(DP_BRUSH_BKGD_CLR);
877 }
878 const uint32_t d = pixel_read(di, bpp);
879 pixel_write(
880 di, bpp, (rop3(op, pattern, s, d) & mask) | (d & ~mask));
881 }
882}

884// Zero-width Bresenham line from DST_LINE_START, excluding the end pixel (XAA
885// draws it separately). Solid and 32x1 mono brushes; the brush phase is
886// DST_LINE_PATCOUNT[4:0].
887void CRV200::draw_line(uint32_t start, uint32_t end)
888{
889 if (m_pending.active)
890 {
891 fail("new primitive while host upload is pending");
892 return;
893 }
894 const uint32_t g = r(DP_GUI_MASTER_CNTL), type = r(DP_DATATYPE),
895 bpp = bytes_per_pixel(type & 15), brush = (type >> 8) & 15;
896 const uint8_t op = uint8_t(r(DP_MIX) >> 16);
897 const bool uses_p = rop_uses(op, 2);
898 if (!bpp)
899 {
900 fail("unsupported pixel format");
901 return;
902 }
903 if ((g & (1u << 27)) || r(CLR_CMP_CNTL))
904 {
905 fail("3D or color comparison in line operation");
906 return;
907 }
908 if (rop_uses(op, 1))
909 {
910 fail("source-dependent line ROP unsupported");
911 return;
912 }
913 if (uses_p && brush != 13 && brush != 6 && brush != 7)
914 {
915 fail("unsupported line brush datatype");
916 return;
917 }
918 const uint32_t dst = r(DST_PITCH_OFFSET);
919 if (dst & 0xc0000000u)
920 {
921 fail("tiled engine surfaces unsupported");
922 return;
923 }
924 const uint32_t db = (dst & 0x3fffffu) << 10, dp = (dst >> 16) & 0x3fc0u;
925 const int32_t x0 = int16_t(start & 0xffff), y0 = int16_t(start >> 16);
926 const int32_t dx = int16_t(end & 0xffff) - x0, dy = int16_t(end >> 16) - y0;
927 const int32_t adx = dx < 0 ? -dx : dx, ady = dy < 0 ? -dy : dy;
928 const int32_t sx = dx < 0 ? -1 : 1, sy = dy < 0 ? -1 : 1;
929 const bool xmajor = adx >= ady;
930 const int32_t major = xmajor ? adx : ady, minor = xmajor ? ady : adx;
931 const uint32_t tl = r(SC_TOP_LEFT), br = r(SC_BOTTOM_RIGHT),
932 mask = r(DP_WRITE_MSK), fg = r(DP_BRUSH_FRGD_CLR),
933 bg = r(DP_BRUSH_BKGD_CLR), pattern = r(BRUSH_DATA0),
934 phase = r(DST_LINE_PATCOUNT) & 31;
935 const bool lsb = (type & (1u << 30)) != 0;
936 // Validate the unclipped footprint before modifying VRAM.
937 for (int pass = 0; pass < 2; ++pass)
938 {
939 int32_t x = x0, y = y0, err = 2 * minor - major;
940 for (int32_t i = 0; i < major; ++i)
941 {
942 size_t di = 0;
943 if (!clipped(x, y, tl, br))
944 {
945 if (!pixel_address(db, dp, x, y, bpp, di))
946 {
947 fail("line address outside GPU framebuffer window/pitch");
948 return;
949 }
950 const bool ink =
951 brush == 13 || mono_bit(pattern, (phase + i) & 31, lsb);
952 if (pass && (ink || brush != 7 || !uses_p))
953 {
954 const uint32_t d = pixel_read(di, bpp);
955 const uint32_t p = ink ? fg : bg;
956 pixel_write(
957 di, bpp, (rop3(op, p, 0, d) & mask) | (d & ~mask));
958 }
959 }
960 if (err > 0)
961 {
962 if (xmajor)
963 y += sy;
964 else
965 x += sx;
966 err -= 2 * major;
967 }
968 err += 2 * minor;
969 if (xmajor)
970 x += sx;
971 else
972 y += sy;
973 }
974 }
975 m_counters.draws++;
976}

978void CRV200::host_word(uint32_t value, bool last)
979{
980 if (halted())
981 return;
982 if (!m_pending.active)
983 {
984 fail("HOST_DATA without pending upload");
985 return;
986 }
987 auto& p = m_pending;
988 const bool mono = p.source_format < 2;
989 const uint32_t pixels_per_word = mono ? 32u : 4u / p.bpp;
990 const uint32_t row = p.word / p.words_per_row,
991 col = (p.word % p.words_per_row) * pixels_per_word;
992 if (last && p.word + 1 != p.words_per_row * p.height)
993 {
994 fail("premature HOST_DATA_LAST");
995 return;
996 }
997 for (uint32_t n = 0; n < pixels_per_word && col + n < p.width; n++)
998 {
999 const int32_t x = p.x + p.dx * int32_t(col + n),
1000 y = p.y + p.dy * int32_t(row);
1001 size_t idx = 0;
1002 if (clipped(x, y, p.sc_tl, p.sc_br))
1003 continue;
1004 if (!pixel_address(p.base, p.pitch, x, y, p.bpp, idx))
1005 {
1006 fail("host upload address invalidated");
1007 return;
1008 }
1009 uint32_t source;
1010 if (mono)
1011 {
1012 const bool ink = mono_bit(value, n, p.mono_lsb != 0);
1013 if (!ink && p.source_format == 1)
1014 continue;
1015 source = ink ? p.fg : p.bg;
1016 }
1017 else
1018 source = value >> (8 * p.bpp * n);
1019 if (source_rejected(
1020 p.compare, p.compare_color, p.compare_mask, source, p.bpp))
1021 continue;
1022 const uint32_t d = pixel_read(idx, p.bpp);
1023 pixel_write(
1024 idx, p.bpp,
1025 (rop3(uint8_t(p.rop), p.pattern, source, d) & p.mask) |
1026 (d & ~p.mask));
1027 }
1028 if (++p.word == p.words_per_row * p.height)
1029 p.active = false;
1030}

1034void CRV200::advance_scanlines(uint32_t lines)
1035{
1036 // Register-derived vertical timing; PLL frequency is not evaluated.
1037 if (lines > 1000000u)
1038 {
1039 fail("scanline budget exceeded");
1040 return;
1041 }
1042 // An atomic PLL update completes on the next model step.
1043 m_pll[PPLL_REF_DIV] &= ~(1u << 15);
1044 if (!(r(CRTC_GEN_CNTL) & CRTC_EN))
1045 return;
1046 const uint32_t total = (r(CRTC_V_TOTAL_DISP) & 0xfff) + 1,
1047 display = ((r(CRTC_V_TOTAL_DISP) >> 16) & 0xfff) + 1;
1048 if (display >= total)
1049 return;
1050 for (uint32_t n = 0; n < lines; n++)
1051 {
1052 if (++m_scanline >= total)
1053 {
1054 m_scanline = 0;
1055 m_frame++;
1056 }
1057 if (m_scanline == display)
1058 {
1059 r(GEN_INT_STATUS) |= 1;
1060 r(CRTC_STATUS) |= CRTC_VBLANK_SAVE;
1061 }
1062 }
1063 update_irq();
1064}

1066CRV200::Frame CRV200::scanout(std::string* error) const
1067{
1068 auto reject = [error](const char* why) {
1069 if (error)
1070 *error = why;
1071 return Frame{};
1072 };
1073 const uint32_t gen = r(CRTC_GEN_CNTL);
1074 if (!(gen & CRTC_EN) || (r(CRTC_EXT_CNTL) & DISPLAY_DIS))
1075 return reject("CRTC disabled/blanked");
1076 if (gen & 3u)
1077 return reject("interlaced/doublescan scanout unsupported");
1078 if (r(CRTC_OFFSET_CNTL))
1079 return reject(
1080 "nonzero CRTC_OFFSET_CNTL requires an unimplemented display "
1081 "layout");
1082 const uint32_t fmt = (gen >> 8) & 15, bpp = bytes_per_pixel(fmt);
1083 const uint32_t w = (((r(CRTC_H_TOTAL_DISP) >> 16) & 0x1ff) + 1) * 8,
1084 h = ((r(CRTC_V_TOTAL_DISP) >> 16) & 0xfff) + 1;
1085 const uint32_t pitch = (r(CRTC_PITCH) & 0x7ff) * 8 * bpp,
1086 base = r(CRTC_OFFSET) & ~7u;
1087 if (!bpp || w > 4096 || h > 4096 || uint64_t(w) * bpp > pitch)
1088 return reject("unsupported/invalid display format or pitch");
1089 size_t first = 0, last = 0;
1090 if (!translate(base, bpp, first) ||
1091 !translate(
1092 uint64_t(base) + uint64_t(h - 1) * pitch + uint64_t(w - 1) * bpp,
1093 bpp, last))
1094 return reject("display address outside GPU framebuffer window");
1095 Frame f;
1096 f.width = w;
1097 f.height = h;
1098 f.argb.resize(size_t(w) * h);
1099 for (uint32_t y = 0; y < h; y++)
1100 for (uint32_t x = 0; x < w; x++)
1101 {
1102 uint32_t v = pixel_read(
1103 first + size_t(y) * pitch + size_t(x) * bpp, bpp),
1104 rr = 0, gg = 0, bb = 0;
1105 if (fmt == 2)
1106 {
1107 v = m_palette[v & 255];
1108 rr = (v >> 16) & 255;
1109 gg = (v >> 8) & 255;
1110 bb = v & 255;
1111 }
1112 else if (fmt == 3 || fmt == 4)
1113 {
1114 rr = (v >> (fmt == 3 ? 10 : 11)) & 31;
1115 gg = (v >> 5) & (fmt == 3 ? 31 : 63);
1116 bb = v & 31;
1117 rr = (rr << 3) | (rr >> 2);
1118 bb = (bb << 3) | (bb >> 2);
1119 gg = fmt == 3 ? ((gg << 3) | (gg >> 2))
1120 : ((gg << 2) | (gg >> 4));
1121 }
1122 else
1123 {
1124 rr = (v >> 16) & 255;
1125 gg = (v >> 8) & 255;
1126 bb = v & 255;
1127 }
1128 // True-color LUT/gamma is bypassed by this diagnostic presenter.
1129 f.argb[size_t(y) * w + x] =
1130 0xff000000u | (rr << 16) | (gg << 8) | bb;
1131 }
1132 composite_cursor(f);
1133 return f;
1134}

1136// 64x64 cursor at CUR_OFFSET. Mono rows: 8 AND then 8 XOR bytes, MSB first
1137// (XFree86 R128/Radeon); ARGB rows: 64 premultiplied pixels. HORZ_VERT_OFF
1138// skips left columns and shortens the top.
1139void CRV200::composite_cursor(Frame& f) const
1140{
1141 const uint32_t gen = r(CRTC_GEN_CNTL), mode = (gen >> 20) & 7;
1142 if (!(gen & CRTC_CUR_EN) || (mode != 0 && mode != 2))
1143 return;
1144 const uint32_t posn = r(CUR_HORZ_VERT_POSN), off = r(CUR_HORZ_VERT_OFF);
1145 const uint32_t px = (posn >> 16) & 0x3fff, py = posn & 0x3fff;
1146 const uint32_t xoff = (off >> 16) & 0x3f, yoff = off & 0x3f;
1147 const uint32_t stride = mode ? 256 : 16;
1148 const uint32_t clr0 = r(CUR_CLR0) & 0xffffff, clr1 = r(CUR_CLR1) & 0xffffff;
1149 for (uint32_t row = 0; row < 64 - yoff && py + row < f.height; ++row)
1150 {
1151 size_t line = 0;
1152 if (!translate(
1153 uint64_t(r(CUR_OFFSET)) + uint64_t(row) * stride, stride, line))
1154 return;
1155 for (uint32_t col = xoff; col < 64 && px + col - xoff < f.width; ++col)
1156 {
1157 uint32_t& d = f.argb[size_t(py + row) * f.width + px + col - xoff];
1158 if (!mode)
1159 {
1160 const unsigned bit = 7 - (col & 7);
1161 const bool and_bit = (m_vram[line + col / 8] >> bit) & 1;
1162 const bool xor_bit = (m_vram[line + 8 + col / 8] >> bit) & 1;
1163 if (!and_bit)
1164 d = 0xff000000u | (xor_bit ? clr1 : clr0);
1165 else if (xor_bit)
1166 d ^= 0x00ffffffu;
1167 continue;
1168 }
1169 const uint32_t c = pixel_read(line + col * 4, 4), a = c >> 24;
1170 uint32_t out = 0xff000000u;
1171 for (unsigned shift = 0; shift < 24; shift += 8)
1172 {
1173 const uint32_t v = ((c >> shift) & 255) +
1174 (((d >> shift) & 255) * (255 - a) + 127) / 255;
1175 out |= std::min(v, 255u) << shift;
1176 }
1177 d = out;
1178 }
1179 }
1180}

1210void CRV200::reset(bool clear_vram)
1211{
1212 // PCI configuration belongs to the board wrapper and is NOT reset here.
1213 m_regs.fill(0);
1214 m_pll.fill(0);
1215 m_warned.fill(false);
1216 m_aperture_warned = false;
1217 m_fault.clear();
1218 m_pending = Pending{};

1284{

1287 {
1288 c ^= v;
1289 for (int i = 0; i < 8; i++)

1302{
