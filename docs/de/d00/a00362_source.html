326uint8_t CVGA::vga_latch_write(int offs, uint8_t data)
327{
328 uint8_t res = 0;
329
330 switch (vga.gc.write_mode & 3)
331 {
332 case 0:
333 data = rotate_right(data);
334 if (vga.gc.enable_set_reset & 1 << offs)
335 res = vga_logical_op((vga.gc.set_reset & 1 << offs) ? vga.gc.bit_mask : 0, offs, vga.gc.bit_mask);
336 else
337 res = vga_logical_op(data, offs, vga.gc.bit_mask);
338 break;
339 case 1:
340 res = vga.gc.latch[offs];
341 break;
342 case 2:
343 res = vga_logical_op((data & 1 << offs) ? 0xff : 0x00, offs, vga.gc.bit_mask);
344 break;
345 case 3:
346 data = rotate_right(data);
347 res = vga_logical_op((vga.gc.set_reset & 1 << offs) ? 0xff : 0x00, offs, data & vga.gc.bit_mask);
348 break;
349 }
350
351 return res;
352}

354void CVGA::vga_vh_text(bitmap_rgb32& bitmap, const rectangle& cliprect)
355{
356 // TODO: make it a subclassable function, cfr. PR2 Video Select in Paradise family
357#define TEXT_COPY_9COLUMN(ch) (((ch & 0xe0) == 0xc0) && (vga.attribute.data[0x10] & 4))
358
359 int width = BIT(vga.sequencer.data[1], 0) ? 8 : 9, height = (vga.crtc.maximum_scan_line) * (vga.crtc.scan_doubling + 1);
360
361 const u32 frame_number = (u32)(screen().frame_number());
362
363 if (vga.crtc.cursor_enable)
364 vga.cursor.visible = frame_number & 0x10;
365 else
366 vga.cursor.visible = 0;
367
368 bool blink_rate = frame_number & 0x20;
369 const int TEXT_LINES = vga.crtc.vert_disp_end + 1;
370
371 unsigned addr = 0;
372 int line = 0;
373 for (addr = vga.crtc.start_addr, line = -vga.crtc.preset_row_scan; line < TEXT_LINES;
374 line += height, addr += (offset() >> 1))
375 {
376 for (unsigned pos = addr, column = 0; column < vga.crtc.horz_disp_end + 1; column++, pos++)
377 {
378 const uint8_t ch = vga.memory[(pos << 1) + 0];
379 const uint8_t attr = vga.memory[(pos << 1) + 1];
380 const uint32_t font_base = vga.sequencer.char_sel.base[BIT(attr, 3)] + (ch << 5);
381 const bool blink_sel = !!BIT(vga.attribute.data[0x10], 3);
382 const uint8_t blink_en = (blink_sel && blink_rate) ? BIT(attr, 7) : 0;
383
384 uint8_t fore_col = attr & 0xf;
385 uint8_t back_col = (attr & 0x70) >> 4;
386 // blink disabled translates MSB of attribute as extra intensity for background color
387 back_col |= (blink_sel) ? 0 : (BIT(attr, 7) >> 4);
388
389 for (int h = std::max(-line, 0); (h < height) && (line + h < std::min(TEXT_LINES, bitmap.height())); h++)
390 {
391 uint32_t* const bitmapline = &bitmap.pix(line + h);
392 uint8_t bits = vga.memory[font_base + (h >> (vga.crtc.scan_doubling))];
393
394 int mask, w;
395 for (mask = 0x80, w = 0; (w < width) && (w < 8); w++, mask >>= 1)
396 {
397 pen_t pen;
398 if (bits & mask)
399 pen = vga.pens[blink_en ? back_col : fore_col];
400 else
401 pen = vga.pens[back_col];
402
403 const int dest_x = column * width + w;
404 if (!screen().visible_area().contains(dest_x, line + h))
405 continue;
406 bitmapline[dest_x] = pen;
407
408 }
409 if (w < width)
410 {
411 /* 9 column */
412 pen_t pen;
413 if (TEXT_COPY_9COLUMN(ch) && (bits & 1))
414 pen = vga.pens[blink_en ? back_col : fore_col];
415 else
416 pen = vga.pens[back_col];
417
418 const int dest_x = column * width + w;
419
420 if (!screen().visible_area().contains(dest_x, line + h))
421 continue;
422 bitmapline[dest_x] = pen;
423 }
424 }
425 if (vga.cursor.visible && (pos == vga.crtc.cursor_addr))
426 {
427 for (int h = vga.crtc.cursor_scan_start;
428 (h <= vga.crtc.cursor_scan_end) && (h < height) && (line + h < TEXT_LINES);
429 h++)
430 {
431 const int dest_x = column * width;
432
433 if (!screen().visible_area().contains(dest_x, line + h))
434 continue;
435 bitmap.plot_box(dest_x, line + h, width, 1, vga.pens[attr & 0xf]);
436 }
437 }
438 }
439 }
440}

442void CVGA::vga_vh_ega(bitmap_rgb32& bitmap, const rectangle& cliprect)
443{
444 const int height = vga.crtc.maximum_scan_line * (vga.crtc.scan_doubling + 1);
445 const int pel_shift = (vga.attribute.pel_shift & 7);
446 const int LINES = (vga.crtc.vert_disp_end + 1) * (get_interlace_mode() + 1);
447 const int EGA_COLUMNS = vga.crtc.horz_disp_end + 1;
448
449 for (int addr = vga.crtc.start_addr, line = 0; line < LINES; line += height, addr += offset())
450 {
451 for (int yi = 0; yi < height; yi++)
452 {
453 uint32_t* const bitmapline = &bitmap.pix(line + yi);
454 // ibm_5150:batmanmv uses this on gameplay for both EGA and "VGA" modes
455 // NB: EGA mode in that game sets 663, should be 303 like the other mode
456 // causing no status bar to appear. This is a known btanb in how VGA
457 // handles EGA mode, cfr. https://www.os2museum.com/wp/fantasyland-on-vga/
458 if ((line + yi) == (vga.crtc.line_compare & 0x3ff))
459 addr = 0;
460
461 for (int pos = addr, c = 0, column = 0; column < EGA_COLUMNS + 1; column++, c += 8, pos = (pos + 1) & 0xffff)
462 {
463 int data[4] = {
464 vga.memory[(pos & 0xffff)],
465 vga.memory[(pos & 0xffff) + 0x10000] << 1,
466 vga.memory[(pos & 0xffff) + 0x20000] << 2,
467 vga.memory[(pos & 0xffff) + 0x30000] << 3 };
468
469 for (int i = 7; i >= 0; i--)
470 {
471 pen_t pen = vga.pens[(data[0] & 1) | (data[1] & 2) | (data[2] & 4) | (data[3] & 8)];
472
473 data[0] >>= 1;
474 data[1] >>= 1;
475 data[2] >>= 1;
476 data[3] >>= 1;
477
478 if (!screen().visible_area().contains(c + i - pel_shift, line + yi))
479 continue;
480 bitmapline[c + i - pel_shift] = pen;
481 }
482 }
483 }
484 }
485}

487// In mode 13h (256 colors) every pixel actually double height/width
488// i.e. a 320x200 is really 640x400
489void CVGA::vga_vh_vga(bitmap_rgb32& bitmap, const rectangle& cliprect)
490{
491 int height = vga.crtc.maximum_scan_line * (vga.crtc.scan_doubling + 1);
492 int pel_shift = (vga.attribute.pel_shift & 6);
493 int addrmask = vga.crtc.no_wrap ? -1 : 0xffff;
494
495 const int LINES = (vga.crtc.vert_disp_end + 1) * (get_interlace_mode() + 1);
496 /* line compare is screen sensitive */
497 uint16_t mask_comp = 0x3ff; //| (LINES & 0x300);
498 const int VGA_COLUMNS = vga.crtc.horz_disp_end + 1;
499 // popmessage("%02x %02x",vga.attribute.pel_shift,vga.sequencer.data[4] & 0x08);
500
501 int curr_addr = 0;
502 if (!(vga.sequencer.data[4] & 0x08))
503 {
504 for (int addr = start_addr(), line = 0; line < LINES; line += height, addr += offset(), curr_addr += offset())
505 {
506 for (int yi = 0; yi < height; yi++)
507 {
508 if ((line + yi) < (vga.crtc.line_compare & mask_comp))
509 curr_addr = addr;
510 if ((line + yi) == (vga.crtc.line_compare & mask_comp))
511 {
512 curr_addr = 0;
513 pel_shift = 0;
514 }
515 uint32_t* const bitmapline = &bitmap.pix(line + yi);
516 for (int pos = curr_addr, c = 0, column = 0; column < VGA_COLUMNS + 1; column++, c += 8, pos++)
517 {
518 if (pos > 0x80000 / 4)
519 return;
520
521 for (int xi = 0; xi < 8; xi++)
522 {
523 if (!screen().visible_area().contains(c + xi - (pel_shift), line + yi))
524 continue;
525 bitmapline[c + xi - (pel_shift)] = pen(vga.memory[(pos & addrmask) + ((xi >> 1) * 0x10000)]);
526 }
527 }
528 }
529 }
530 }
531 else
532 {
533 for (int addr = start_addr(), line = 0; line < LINES; line += height, addr += offset(), curr_addr += offset())
534 {
535 for (int yi = 0; yi < height; yi++)
536 {
537 if ((line + yi) < (vga.crtc.line_compare & mask_comp))
538 curr_addr = addr;
539 if ((line + yi) == (vga.crtc.line_compare & mask_comp))
540 curr_addr = 0;
541 uint32_t* const bitmapline = &bitmap.pix(line + yi);
542 //addr %= 0x80000;
543 for (int pos = curr_addr, c = 0, column = 0; column < VGA_COLUMNS + 1; column++, c += 0x10, pos += 0x8)
544 {
545 if (pos + 0x08 > 0x80000)
546 return;
547
548 for (int xi = 0; xi < 0x10; xi++)
549 {
550 if (!screen().visible_area().contains(c + xi - (pel_shift), line + yi))
551 continue;
552 bitmapline[c + xi - pel_shift] = pen(vga.memory[(pos + (xi >> 1)) & addrmask]);
553 }
554 }
555 }
556 }
557 }
558}

560void CVGA::vga_vh_cga(bitmap_rgb32& bitmap, const rectangle& cliprect)
561{
562 const int LINES = (vga.crtc.vert_disp_end + 1) * (get_interlace_mode() + 1);
563
564 const int height = (vga.crtc.scan_doubling + 1);
565 const int width = (vga.crtc.horz_disp_end + 1) * 8;
566
567 for (int y = 0; y < LINES; y++)
568 {
569 uint32_t addr = ((y & 1) * 0x2000) + (((y & ~1) >> 1) * width / 4);
570
571 for (int x = 0; x < width; x += 4)
572 {
573 for (int yi = 0; yi < height; yi++)
574 {
575 uint32_t* const bitmapline = &bitmap.pix(y * height + yi);
576
577 for (int xi = 0; xi < 4; xi++)
578 {
579 pen_t pen = vga.pens[(vga.memory[addr] >> (6 - xi * 2)) & 3];
580 if (!screen().visible_area().contains(x + xi, y * height + yi))
581 continue;
582 bitmapline[x + xi] = pen;
583 }
584 }
585
586 addr++;
587 }
588 }
589}

855uint8_t CVGA::pc_vga_choosevideomode()
856{
857 if (vga.crtc.sync_en)
858 {
859 if (vga.dac.dirty)
860 {
861 palette_update();
862 vga.dac.dirty = 0;
863 }
864
865 if (vga.attribute.data[0x10] & 0x80)
866 {
867 for (int i = 0; i < 16; i++)
868 {
869 vga.pens[i] = pen((vga.attribute.data[i] & 0x0f)
870 | ((vga.attribute.data[0x14] & 0xf) << 4));
871 }
872 }
873 else
874 {
875 for (int i = 0; i < 16; i++)
876 {
877 vga.pens[i] = pen((vga.attribute.data[i] & 0x3f)
878 | ((vga.attribute.data[0x14] & 0xc) << 4));
879 }
880 }
881
882 if (svga.rgb32_en)
883 {
884 return RGB32_MODE;
885 }
886 else if (svga.rgb24_en)
887 {
888 return RGB24_MODE;
889 }
890 else if (svga.rgb16_en)
891 {
892 return RGB16_MODE;
893 }
894 else if (svga.rgb15_en)
895 {
896 return RGB15_MODE;
897 }
898 else if (svga.rgb8_en)
899 {
900 return RGB8_MODE;
901 }
902 else if (!vga.gc.alpha_dis) // !GRAPHIC_MODE
903 {
904 return TEXT_MODE;
905 }
906 else if (vga.gc.shift256)
907 {
908 return VGA_MODE;
909 }
910 else if (vga.gc.shift_reg)
911 {
912 return CGA_MODE;
913 }
914 else if (vga.gc.memory_map_sel == 0x03)
915 {
916 return MONO_MODE;
917 }
918 else
919 {
920 return EGA_MODE;
921 }
922 }
923
924 return SCREEN_OFF;
925}

1072 if ((offset & 0x18000) != 0x10000)
1073 return;
1074
