198 inline uint8_t vga_logical_op(uint8_t data, uint8_t plane, uint8_t mask)
199 {
200 uint8_t res = 0;
201 switch (vga.gc.logical_op & 3)
202 {
203 case 0: /* NONE */
204 res = (data & mask) | (vga.gc.latch[plane] & ~mask);
205 break;
206 case 1: /* AND */
207 res = (data | ~mask) & (vga.gc.latch[plane]);
208 break;
209 case 2: /* OR */
210 res = (data & mask) | (vga.gc.latch[plane]);
211 break;
212 case 3: /* XOR */
213 res = (data & mask) ^ (vga.gc.latch[plane]);
214 break;
215 }
216 return res;
217 }
