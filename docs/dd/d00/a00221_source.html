39 static constexpr AlphaIPRReadDescriptor describe(std::uint32_t function)
40 {
41 if ((function & 0xc0) == 0x40)
42 return {AlphaIPRReadKind::Helper, false, true}; // PCTX group.
43
44 switch (function)
45 {
46 case 0x06: return {AlphaIPRReadKind::ExceptionAddress, false, true};
47 case 0x0f: return {AlphaIPRReadKind::ExceptionSummary, false, true};
48 case 0x10: return {AlphaIPRReadKind::PalBase, false, true};
49 case 0x27: return {AlphaIPRReadKind::MemoryFaultStatus, false, true};
50 case 0xc2: return {AlphaIPRReadKind::FaultAddress, false, true};
51 case 0x0d: return {AlphaIPRReadKind::Helper, true, true}; // ISUM is asynchronous.
52 case 0x05: case 0x07:
53 case 0x08: case 0x09: case 0x0a: case 0x0b: case 0x0c:
54 case 0x11: case 0x14: case 0x16: case 0x2a: case 0x2b:
55 case 0xc0: case 0xc3:
56 return {AlphaIPRReadKind::Helper, false, true};
57 default:
58 return {AlphaIPRReadKind::Unsupported, false, true};
59 }
60 }
61
62 static constexpr AlphaIPRWriteKind describe_write(std::uint32_t function)
63 {
64 // ASN changes require interpreter dispatch; the other PCTX fields can compile.
65 if ((function & 0xc0) == 0x40)
66 return (function & 1) ? AlphaIPRWriteKind::Interpreter : AlphaIPRWriteKind::Helper;
67
68 switch (function)
69 {
70 case 0x00: case 0x14: case 0x20: case 0x26:
71 case 0x29: case 0xa0: case 0xc0: // Field stores.
72 case 0x01: case 0x21: case 0xa1: // Translation-buffer fills.
73 case 0x02: case 0x03: case 0x04: // ITB invalidation.
74 case 0x13: // Lazy instruction-cache flush.
75 case 0x09: case 0x0a: case 0x0b: case 0x0c: // Mode/interrupt fields.
76 case 0x24: case 0xa4: // Single-entry DTB invalidation.
77 return AlphaIPRWriteKind::Helper;
78 case 0x11: // I_CTL changes compiled assumptions, including SDE.
79 return AlphaIPRWriteKind::HelperTerminator;
80 case 0x15: case 0x17: case 0x27:
81 case 0x2b: case 0x2c: case 0x2d:
82 return AlphaIPRWriteKind::Noop;
83 default:
84 return AlphaIPRWriteKind::Interpreter;
85 }
86 }
