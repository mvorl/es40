67// Decoded PAL return rules, outside saved CPU state. The target uses the live
68// source register; return forms with side effects stay interpreted.
69struct AlphaPALReturnDescriptor
70{
71 bool valid = false;
72 std::uint64_t target_mask = ~std::uint64_t(0);
73 std::uint64_t target_bits = 0;
74 bool clear_lock = false;
75 bool compilable = false;
76};
77
78constexpr std::uint64_t alpha_pal_return_target(
79 const AlphaPALReturnDescriptor& entry, std::uint64_t source)
80{
81 return (source & entry.target_mask) | entry.target_bits;
82}
83
84constexpr bool alpha_pal_return_compilable(const AlphaPALReturnDescriptor& entry)
85{
86 return entry.valid && entry.compilable && !entry.clear_lock;
87}

89// Family access rules for the PAL-reserved hardware instructions.
90enum class AlphaPALInstructionAccess
91{
92 Denied,
93 PalOnly,
94 PalOrKernelHwe
95};
96
97constexpr bool alpha_pal_instruction_allowed(AlphaPALInstructionAccess access,
98 bool pal_mode, std::uint32_t mode,
99 bool hwe)
100{
101 switch (access)
102 {
103 case AlphaPALInstructionAccess::Denied: return false;
104 case AlphaPALInstructionAccess::PalOnly: return pal_mode;
105 case AlphaPALInstructionAccess::PalOrKernelHwe:
106 return pal_mode || (mode == 0 && hwe);
107 }
108 return false;
109}
110
111// These rules guarantee access throughout a PAL block.
112// Native HWE execution stays interpreted.
113constexpr bool alpha_pal_instruction_compilable(AlphaPALInstructionAccess access)
114{
115 return access == AlphaPALInstructionAccess::PalOnly
116 || access == AlphaPALInstructionAccess::PalOrKernelHwe;
117}

119// Stateless family PAL rules, copied with the CPU profile.
120struct AlphaPALPolicy
121{
122 AlphaPALExceptionEntry (*exception_entry)(AlphaPALException reason,
123 std::uint64_t current_pc);
124 bool (*call_pal_valid)(std::uint32_t function, std::uint32_t mode);
125 AlphaPALCallEntry (*call_pal_entry)(std::uint32_t function,
126 std::uint64_t current_pc,
127 std::uint64_t next_pc);
128 // Pass the complete instruction so restrictions on individual forms apply.
129 // An absent access rule denies all.
130 AlphaPALInstructionAccess (*hardware_instruction_access)(std::uint32_t instruction) = nullptr;
131 AlphaPALReturnDescriptor (*hardware_return)(std::uint32_t instruction) = nullptr;
132};
133
134constexpr AlphaPALReturnDescriptor alpha_pal_return_descriptor(
135 const AlphaPALPolicy& policy, std::uint32_t instruction)
136{
137 return policy.hardware_return ? policy.hardware_return(instruction)
138 : AlphaPALReturnDescriptor{};
139}
