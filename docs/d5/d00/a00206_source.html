101// Family decoding feeds one shared load executor. The caller handles faults
102// and unknown forms in the interpreter instruction-exit context.
103#define DO_HW_LD do { \
104 if (!alpha_hw_load_valid(hw_load)) { UNKNOWN2; } \
105 const u64 hw_address = state.r[REG_2] + (u64)hw_load.displacement; \
106 if (hw_load.address == AlphaHWLoadAddress::Virtual) { \
107 const int hw_flags = ACCESS_READ | (hw_load.vpte ? VPTE : 0) \
108 | (hw_load.alternate ? ALT : 0) | (hw_load.write_check ? WRCHK : 0); \
109 if (hw_flags == ACCESS_READ) { \
110 DATA_PHYS_NT(hw_address, ACCESS_READ); \
111 } else if (virt2phys(hw_address, &phys_address, hw_flags, nullptr, ins)) { \
112 ES40_EXECUTE_END(); \
113 } \
114 } else { \
115 phys_address = hw_address; \
116 } \
117 /* Translate before alignment; preserve raw PA for tracing and locking. */ \
118 state.r[REG_1] = READ_PHYS_NT(hw_load.size_bits); \
119 if (hw_load.size_bits == 32) \
120 state.r[REG_1] = sext_u64_32(state.r[REG_1]); \
121 if (hw_load.locked) \
122 cSystem->cpu_lock(state.iProcNum, phys_address, state.r[REG_1]); \
123} while (0)

128#define DO_HW_STL switch(function) \
129 { \
130 case 0: /* longword physical */ \
131 phys_address = state.r[REG_2] + DISP_12; \
132 WRITE_PHYS_NT(state.r[REG_1], 32); \
133 break; \
134 \
135 case 2: /* longword physical conditional */ \
136 phys_address = state.r[REG_2] + DISP_12; \
137 state.r[REG_1] = cSystem->cpu_stx_c(state.iProcNum, phys_address, 32, \
138 state.r[REG_1], dram_ptr, dram_size, this); \
139 break; \
140 \
141 case 4: /* longword virtual (HRM 6.4.1 Table 6-4 TYPE 0102) -- write \
142 * checked against current mode, matching QEMU brokenpipe \
143 * AlphaMMUIdx_Privileged. */ \
144 DATA_PHYS_NT(state.r[REG_2] + DISP_12, ACCESS_WRITE); \
145 WRITE_PHYS_NT(state.r[REG_1], 32); \
146 break; \
147 \
148 case 12: /* longword virtual alt (HRM TYPE 1102) -- write checked \
149 * using DTB_ALT_MODE, matching brokenpipe AlphaMMUIdx_AltMode.*/\
150 DATA_PHYS_NT(state.r[REG_2] + DISP_12, ACCESS_WRITE | ALT); \
151 WRITE_PHYS_NT(state.r[REG_1], 32); \
152 break; \
153 \
154 default: \
155 UNKNOWN2; \
156 }

158#define DO_HW_STQ switch(function) \
159 { \

163 break; \
164 \
165 case 3: /* quadword physical conditional */ \

167 state.r[REG_1] = cSystem->cpu_stx_c(state.iProcNum, phys_address, 64, \
168 state.r[REG_1], dram_ptr, dram_size, this); \
169 break; \
170 \
171 case 5: /* quadword virtual (HRM 6.4.1 Table 6-4 TYPE 0102) -- \
172 * see HW_STL case 4. */ \
173 DATA_PHYS_NT(state.r[REG_2] + DISP_12, ACCESS_WRITE); \
174 WRITE_PHYS_NT(state.r[REG_1], 64); \
175 break; \
176 \
177 case 13: /* quadword virtual alt (HRM TYPE 1102) -- see HW_STL 12. */ \
178 DATA_PHYS_NT(state.r[REG_2] + DISP_12, ACCESS_WRITE | ALT); \
179 WRITE_PHYS_NT(state.r[REG_1], 64); \
180 break; \
181 \
182 default: \
183 UNKNOWN2; \
