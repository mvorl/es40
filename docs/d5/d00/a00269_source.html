120class address_map
121{
122public:
123 explicit address_map(unsigned size = 256)
124 : m_size(size), m_slots(size), m_unmapval(0xFF), m_globalmask(0)
125 {
126 }
127
128 address_map_entry operator()(unsigned lo, unsigned hi)
129 {
130 return address_map_entry(*this, lo, hi);
131 }
132
133 // map-level
134
135 void unmap_value_high() { m_unmapval = 0xFF; }
136 void unmap_value_low() { m_unmapval = 0x00; }
137 void global_mask(offs_t mask) { m_globalmask = mask; }
138
139 // Dispatch
140
141 u8 read_byte(unsigned addr) const
142 {
143 if (m_globalmask) addr &= m_globalmask;
144 if (addr < m_size && m_slots[addr].read_fn)
145 return m_slots[addr].read_fn(static_cast<offs_t>(addr - m_slots[addr].read_base));
146 return m_unmapval;
147 }
148
149 void write_byte(unsigned addr, u8 data)
150 {
151 if (m_globalmask) addr &= m_globalmask;
152 if (addr < m_size && m_slots[addr].write_fn)
153 m_slots[addr].write_fn(static_cast<offs_t>(addr - m_slots[addr].write_base), data);
154 }
155
156 bool has_handler(unsigned addr) const
157 {
158 if (m_globalmask) addr &= m_globalmask;
159 if (addr >= m_size) return false;
160 return m_slots[addr].read_fn || m_slots[addr].write_fn;
161 }
162
163 const char* debug_name(unsigned addr) const
164 {
165 if (m_globalmask) addr &= m_globalmask;
166 return (addr < m_size) ? m_slots[addr].name : nullptr;
167 }
168
169 u8 unmapval() const { return m_unmapval; }
170
171 // called by address_map_entry
172
173 void install_read(unsigned addr, std::function<u8(offs_t)> fn, unsigned base, const char* name)
174 {
175 if (addr < m_size)
176 {
177 m_slots[addr].read_fn = std::move(fn);
178 m_slots[addr].read_base = base;
179 if (name) m_slots[addr].name = name;
180 }
181 }
182
183 void install_write(unsigned addr, std::function<void(offs_t, u8)> fn, unsigned base, const char* name)
184 {
185 if (addr < m_size)
186 {
187 m_slots[addr].write_fn = std::move(fn);
188 m_slots[addr].write_base = base;
189 if (name) m_slots[addr].name = name;
190 }
191 }
192
193 void install_nop(unsigned addr, unsigned base, const char* name)
194 {
195 if (addr < m_size)
196 {
197 m_slots[addr].read_fn = [](offs_t) -> u8 { return 0; };
198 m_slots[addr].write_fn = [](offs_t, u8) {};
199 m_slots[addr].read_base = base;
200 m_slots[addr].write_base = base;
201 if (name) m_slots[addr].name = name;
202 }
203 }
204
205 unsigned size() const { return m_size; }
206
207private:
208 struct Slot
209 {
210 std::function<u8(offs_t)> read_fn;
211 std::function<void(offs_t, u8)> write_fn;
212 unsigned read_base = 0;
213 unsigned write_base = 0;
214 const char* name = nullptr;
215 };
216
217 unsigned m_size;
218 std::vector<Slot> m_slots;
219 u8 m_unmapval;
220 offs_t m_globalmask;
221};

224// mirror expansion helper
225
226inline void address_map_entry::for_each_addr(std::function<void(unsigned)> fn) const
227{
228 if (m_mirror == 0)
229 {
230 for (unsigned a = m_lo; a <= m_hi; ++a)
231 fn(a);
232 }
233 else
234 {
235 // combinations of mirror bits
236 // any mirror bit can be 0 or 1 independently... i think
237 std::vector<unsigned> mirror_bits;
238 for (unsigned b = 0; b < 32; ++b)
239 if (m_mirror & (1u << b))
240 mirror_bits.push_back(1u << b);
241
242 unsigned combos = 1u << mirror_bits.size();
243 for (unsigned c = 0; c < combos; ++c)
244 {
245 unsigned mirror_val = 0;
246 for (unsigned i = 0; i < mirror_bits.size(); ++i)
247 if (c & (1u << i))
248 mirror_val |= mirror_bits[i];
249
250 for (unsigned a = m_lo; a <= m_hi; ++a)
251 fn(a | mirror_val);
252 }
253 }
254}

259template <typename T, typename U>
260address_map_entry& address_map_entry::lrw8(T&& read_fn, const char* read_name,
261 U&& write_fn, const char* write_name)
262{
263 auto rfn = std::make_shared<std::function<u8(offs_t)>>(std::forward<T>(read_fn));
264 auto wfn = std::make_shared<std::function<void(offs_t, u8)>>(std::forward<U>(write_fn));
265 unsigned base = m_lo;
266
267 for_each_addr([&](unsigned addr) {
268 m_map.install_read(addr, [rfn](offs_t off) { return (*rfn)(off); }, base, read_name);
269 m_map.install_write(addr, [wfn](offs_t off, u8 d) { (*wfn)(off, d); }, base, write_name);
270 });
271 return *this;
272}
