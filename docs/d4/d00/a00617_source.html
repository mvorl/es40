601 FAILURE_4(Runtime, "Display outputs %s/%u and %s/%u share a GUI",
602 previous.component->devid_string, previous.output->id(),
603 component->devid_string, output->id());
604 }
605 outputs.push_back(binding);
606 }

714/*
715 * AlphaBIOS does not create an EBDA. Reserve one KiB at the conventional
716 * memory boundary before an x86 option ROM reads BAR6.
717 */
718void CSystem::PrepareX86OptionROMMemory()
719{
720 static const u64 BDA_EBDA_SEGMENT = U64(0x0000040e);
721 static const u64 BDA_MEMORY_SIZE = U64(0x00000413);
722 static const u64 BDA_TIMER_TICKS = U64(0x0000046c);
723 static const u32 ALPHABIOS_IVT_ENTRY = 0xfdec0008U;
724 static const u32 ALPHABIOS_IVT_08 = 0xfdec0020U;
725 static const u32 MAXIMUM_CONVENTIONAL_MEMORY_KB = 576U;
726 static const u32 MINIMUM_CONVENTIONAL_MEMORY_KB = 128U;
727 static const u32 EBDA_BLOCK_SIZE = 1024U;
728 static const u32 EBDA_SIZE_KB = 1U;
729
730 if (iNumMemoryBits < 20 ||
731 ReadMem(0x0000, 32, nullptr) != ALPHABIOS_IVT_ENTRY ||
732 ReadMem(0x0004, 32, nullptr) != ALPHABIOS_IVT_ENTRY ||
733 ReadMem(0x0020, 32, nullptr) != ALPHABIOS_IVT_08)
734 return;
735
736 u32 conventional_memory_kb =
737 (u32)ReadMem(BDA_MEMORY_SIZE, 16, nullptr);
738 if (conventional_memory_kb < MINIMUM_CONVENTIONAL_MEMORY_KB ||
739 conventional_memory_kb > MAXIMUM_CONVENTIONAL_MEMORY_KB)
740 return;
741
742 if (ReadMem(BDA_EBDA_SEGMENT, 16, nullptr) == 0)
743 {
744 const u64 ebda_address = (u64)conventional_memory_kb * 1024U;
745 for (u64 address = ebda_address;
746 address < ebda_address + EBDA_BLOCK_SIZE;
747 ++address)
748 {
749 if (ReadMem(address, 8, nullptr) != 0)
750 return;
751 }
752
753 WriteMem(BDA_EBDA_SEGMENT, 16, ebda_address >> 4, nullptr);
754 WriteMem(ebda_address, 8, EBDA_SIZE_KB, nullptr);
755 }
756
757 if (!m_x86_bios_clock_active.load(std::memory_order_acquire))
758 {
759 m_x86_bios_clock_initial_ticks =
760 (u32)ReadMem(BDA_TIMER_TICKS, 32, nullptr);
761 m_x86_bios_clock_elapsed_ticks = 0;
762 m_x86_bios_clock_epoch = std::chrono::steady_clock::now();
763 m_x86_bios_clock_active.store(true, std::memory_order_release);
764 }
765}

767/* AlphaBIOS's x86 interpreter does not advance the 18.2 Hz BDA clock. */
768void CSystem::UpdateX86BIOSClock()
769{
770 static const u64 BDA_TIMER_TICKS = U64(0x0000046c);
771 static const u64 BDA_TIMER_ROLLOVER = U64(0x00000470);
772 static const u32 ALPHABIOS_IVT_ENTRY = 0xfdec0008U;
773 static const u32 BIOS_TICKS_PER_DAY = 0x001800b0U;
774
775 if (!m_x86_bios_clock_active.load(std::memory_order_acquire))
776 return;
777
778 const u64 elapsed_ms = (u64)std::chrono::duration_cast<
779 std::chrono::milliseconds>(std::chrono::steady_clock::now() -
780 m_x86_bios_clock_epoch).count();
781 const u64 elapsed_ticks = elapsed_ms * 182U / 10000U;
782 if (elapsed_ticks == m_x86_bios_clock_elapsed_ticks)
783 return;
784
785 if (ReadMem(0x0000, 32, nullptr) != ALPHABIOS_IVT_ENTRY ||
786 ReadMem(0x0004, 32, nullptr) != ALPHABIOS_IVT_ENTRY)
787 {
788 m_x86_bios_clock_active.store(false, std::memory_order_release);
789 return;
790 }
791
792 const u64 previous_ticks = m_x86_bios_clock_initial_ticks +
793 m_x86_bios_clock_elapsed_ticks;
794 const u64 current_ticks = m_x86_bios_clock_initial_ticks + elapsed_ticks;
795 WriteMem(BDA_TIMER_TICKS, 32, current_ticks % BIOS_TICKS_PER_DAY, nullptr);
796
797 const u64 rollovers = current_ticks / BIOS_TICKS_PER_DAY -
798 previous_ticks / BIOS_TICKS_PER_DAY;
799 if (rollovers != 0)
800 {
801 const u32 rollover = (u32)ReadMem(BDA_TIMER_ROLLOVER, 8, nullptr);
802 WriteMem(BDA_TIMER_ROLLOVER, 8, rollover + rollovers, nullptr);
803 }
804 m_x86_bios_clock_elapsed_ticks = elapsed_ticks;
805}

1026// EV6/EV68 Dcache blocks are 64 bytes; LDx_L/STx_C monitor that cache line.
1027#define CPU_LOCK_MATCH_MASK U64(0x00000807ffffffc0)
1028#define CPU_LOCK_IO_MASK U64(0x0000080000000000)
1029
1030static inline bool cpu_lock_matches(u64 locked_address, u64 address)
1031{
1032 return !((locked_address ^ address) & CPU_LOCK_MATCH_MASK);
1033}

1035// --- Load-locked / store-conditional (HRM 4.2) -----------------------------
1036// Keep the original CAS-backed model for same-address LL/SC sequences, because
1037// it provides the emulator's MP atomicity. Some Alpha code stores conditionally
1038// to a different quadword in the same locked cache line; those must not compare
1039// against the value loaded from the LDx_L address.
1040
1041void CSystem::cpu_lock(int cpuid, u64 address, u64 value)
1042{
1043 state.cpu_lock_address[cpuid] = address;
1044 cpu_lock_value[cpuid] = value;
1045 // ABA guard - snapshot the line's STx_C sequence.
1046 m_ll_seq_snap[cpuid] =
1047 m_ll_seq[(u32)((address >> 6) & (kLLBuckets - 1))].load(std::memory_order_acquire);
1048 state.cpu_lock_flags |= (1 << cpuid); // atomic fetch_or
1049}
1050
1051bool CSystem::cpu_take_lock(int cpuid, u64 address, u64* expected, bool* same_address)
1052{
1053 // I/O-space conditional stores have no cache line to watch; treat as held.
1054 bool held = (address & CPU_LOCK_IO_MASK) ||
1055 ((state.cpu_lock_flags.load() & (1 << cpuid)) &&
1056 cpu_lock_matches(state.cpu_lock_address[cpuid], address));
1057
1058 // STx_C always consumes this CPU's lock, success or fail.
1059 state.cpu_lock_flags &= ~(1 << cpuid); // atomic fetch_and
1060 if (held)
1061 {
1062 *expected = cpu_lock_value[cpuid];
1063 *same_address = (state.cpu_lock_address[cpuid] == address);
1064 }
1065 return held;
1066}

1071u64 CSystem::cpu_stx_c(int cpuid, u64 phys, int size_bits, u64 value,
1072 char* dram, u64 dram_sz, CSystemComponent* source)
1073{
1074 u64 expected = 0;
1075 bool same_address = false;
1076 if (!cpu_take_lock(cpuid, phys, &expected, &same_address))
1077 return 0;
1078
1079 if (phys >= dram_sz)
1080 {
1081 WriteMem(phys, size_bits, value, source); // I/O-space conditional store
1082 return 1;
1083 }
1084
1085 const u32 b = (u32)((phys >> 6) & (kLLBuckets - 1));
1086 while (m_ll_lk[b].exchange(1, std::memory_order_acquire))
1087 ;
1088 u64 ok;
1089 if (m_ll_seq[b].load(std::memory_order_relaxed) != m_ll_seq_snap[cpuid])
1090 ok = 0; // another STx_C hit this line since our LDx_L
1091 else if (same_address)
1092 ok = dram_cas(dram, phys, expected, value, size_bits) ? 1 : 0;
1093 else
1094 {
1095 // STx_C to a different quadword in the locked line: no value to compare.
1096 dram_write(dram, phys, size_bits, value);
1097 ok = 1;
1098 }
1099 if (ok)
1100 m_ll_seq[b].fetch_add(1, std::memory_order_relaxed);
1101 m_ll_lk[b].store(0, std::memory_order_release);
1102 return ok;
1103}

1144// Called with device_bus_mutex held, before any handler runs.
1145// Positive decode precedes subtractive decode.
1146// A bridge forwarding claim yields to that bridge's specific handler.
1147// Distinct remaining components are claimants.
1148void CSystem::collect_decode_claims(u64 address, int dsize, bool write,
1149 SDecodeClaims& claims, const CSystemComponent* source) const
1150{
1151 claims.count = 0;
1152 claims.subtractive = false;
1153 auto eligible = [&](const SMemoryUser* r) {
1154 return address >= r->base && address < r->base + r->length &&
1155 r->component->decodes_memory_access(r->index, address - r->base, dsize, write);
1156 };
1157
1158 int first = 0;
1159 while (first < iNumMemories && !eligible(asMemories[first].get()))
1160 ++first;
1161 if (first == iNumMemories)
1162 return;
1163 if (!asMemories[first]->may_overlap)
1164 {
1165 claims.range[claims.count++] = *asMemories[first];
1166 return;
1167 }
1168
1169 int positive[SDecodeClaims::kMax * 4];
1170 int subtractive[SDecodeClaims::kMax * 4];
1171 int npos = 0, nsub = 0;
1172 for (int i = first; i < iNumMemories; ++i)
1173 {
1174 const SMemoryUser* r = asMemories[i].get();
1175 if (!eligible(r))
1176 continue;
1177 const bool sub = r->component->uses_subtractive_decode(r->index,
1178 address - r->base, dsize, write);
1179 int* list = sub ? subtractive : positive;
1180 int& n = sub ? nsub : npos;
1181 if (n == SDecodeClaims::kMax * 4)
1182 {
1183 SDecodeClaims partial;
1184 partial.subtractive = sub;
1185 for (int k = 0; k < SDecodeClaims::kMax; ++k)
1186 partial.range[partial.count++] = *asMemories[list[k]];
1187 report_decode_overlap(address, dsize, write, partial, source,
1188 "too many eligible ranges to resolve");
1189 }
1190 list[n++] = i;
1191 }
1192 const int* list = npos ? positive : subtractive;
1193 const int n = npos ? npos : nsub;
1194 claims.subtractive = npos == 0;
1195
1196 for (int k = 0; k < n; ++k)
1197 {
1198 const SMemoryUser* r = asMemories[list[k]].get();
1199 if (r->component->memory_decode_fallback(r->index))
1200 {
1201 bool yields = false;
1202 for (int m = 0; m < n && !yields; ++m)
1203 {
1204 const SMemoryUser* o = asMemories[list[m]].get();
1205 yields = o->component->memory_decode_owner() == r->component->memory_decode_owner() &&
1206 !o->component->memory_decode_fallback(o->index);
1207 }
1208 if (yields)
1209 continue;
1210 }
1211 bool seen = false;
1212 for (int m = 0; m < claims.count && !seen; ++m)
1213 seen = claims.range[m].component == r->component;
1214 if (seen)
1215 continue; // same-component alias
1216 if (claims.count == SDecodeClaims::kMax)
1217 report_decode_overlap(address, dsize, write, claims, source,
1218 "too many claimants to resolve");
1219 claims.range[claims.count++] = *r;
1220 }
1221}

1333// Every claimant performs the read before one value is chosen.
1334// Disagreement is undefined contention on real hardware.
1335u64 CSystem::resolve_shared_read(u64 address, int dsize,
1336 const SDecodeClaims& claims, const CSystemComponent* source)
1337{
1338 const u64 mask = dsize == 64 ? ~U64(0) : (U64(1) << dsize) - 1;
1339 u64 values[SDecodeClaims::kMax];
1340 std::string contexts[SDecodeClaims::kMax];
1341 bool agree = true;
1342 for (int k = 0; k < claims.count; ++k)
1343 {
1344 const SMemoryUser* r = &claims.range[k];
1345 // Capture register context before the read's own side effects.
1346 contexts[k] = r->component->describe_access_context(r->index,
1347 address - r->base);
1348 values[k] = r->component->ReadMem(r->index, address - r->base, dsize) & mask;
1349 agree &= values[k] == values[0];
1350 }
1351 if (agree)
1352 return values[0];
1353
1354 u64 anded = mask, ored = 0, differ = 0;
1355 for (int k = 0; k < claims.count; ++k)
1356 {
1357 anded &= values[k];
1358 ored |= values[k];
1359 differ |= values[k] ^ values[0];
1360 }
1361 std::ostringstream key, detail;
1362 key << std::hex << address << ':' << dsize << ':' << differ;
1363 detail << "claimant values differ, mask 0x" << std::hex << differ
1364 << ", policy ";
1365 switch (shared_read_policy)
1366 {
1367 case SharedReadPolicy::Stop: detail << "stop"; break;
1368 case SharedReadPolicy::Claimant: detail << "claimant " << std::dec
1369 << shared_read_hose << ':' << shared_read_device; break;
1370 case SharedReadPolicy::And: detail << "and"; break;
1371 case SharedReadPolicy::Or: detail << "or"; break;
1372 }
1373 for (int k = 0; k < claims.count; ++k)
1374 {
1375 detail << "\n " << claims.range[k].component->devid_string
1376 << " returned 0x" << std::hex << values[k];
1377 if (!contexts[k].empty())
1378 detail << " [" << contexts[k] << ']';
1379 key << ':' << contexts[k];
1380 }
1381 if (shared_read_policy == SharedReadPolicy::Stop)
1382 report_decode_overlap(address, dsize, false, claims, source,
1383 "claimants returned different read data", detail.str());
1384
1385 std::ostringstream line;
1386 line << "read " << std::dec << dsize << "-bit 0x" << std::hex << address
1387 << " from " << describe_source(source) << ": " << detail.str();
1388 log_shared_event("contention:" + key.str(), line.str());
1389
1390 switch (shared_read_policy)
1391 {
1392 case SharedReadPolicy::And: return anded;
1393 case SharedReadPolicy::Or: return ored;
1394 default: break;
1395 }
1396 for (int k = 0; k < claims.count; ++k)
1397 {
1398 const auto* pci = dynamic_cast<const CPCIDevice*>(
1399 claims.range[k].component);
1400 if (pci && pci->pci_bus() == shared_read_hose &&
1401 pci->pci_dev() == shared_read_device)
1402 return values[k];
1403 }
1404 report_decode_overlap(address, dsize, false, claims, source,
1405 "debug.shared_read_claimant is not among the claimants", detail.str());
1406}

1811void CSystem::SaveState(const char* fn)
1812{
1813 std::lock_guard<std::recursive_mutex> bus_lock(device_bus_mutex);
1814 SDeviceAccessScope time_scope(this);
1815 std::vector<std::string> identities;
1816 for (int i = 0; i < iNumComponents; ++i)
1817 acComponents[i]->prepare_snapshot();
1818 for (int i = 0; i < iNumComponents; ++i)
1819 {
1820 identities.push_back(acComponents[i]->snapshot_identity());
1821 if (identities.back().empty() || identities.back().size() > snapshot_identity_limit)
1822 FAILURE(Runtime, "Invalid snapshot device identity");
1823 }
1824 CSnapshotFile file(fn);
1825 FILE* f = file.get();
1826 const auto write = [f](const void* data, size_t size)
1827 {
1828 if (fwrite(data, size, 1, f) != 1)
1829 FAILURE(Runtime, "Unable to write system state");
1830 };
1831 const u64 memory_size = U64(1) << iNumMemoryBits;
1832 const u32 system_size = (u32)sizeof(state);
1833 const u32 component_count = (u32)iNumComponents;
1834 write(&system_state_magic, sizeof(system_state_magic));
1835 write(&system_state_version, sizeof(system_state_version));
1836 write(&memory_size, sizeof(memory_size));
1837 write(&system_size, sizeof(system_size));
1838 write(&component_count, sizeof(component_count));
1839 for (const auto& identity : identities)
1840 {
1841 const u32 length = (u32)identity.size();
1842 write(&length, sizeof(length));
1843 write(identity.data(), length);
1844 }
1845
1846 const int* mem = (const int*)memory;
1847 const u64 memints = memory_size / sizeof(int);
1848 for (u64 m = 0; m < memints;)
1849 {
1850 const int value = mem[m++];
1851 write(&value, sizeof(value));
1852 if (!value)
1853 {
1854 // A zero word is followed by the number of additional zero words.
1855 u32 extra = 0;
1856 while (m < memints && !mem[m] && extra != UINT32_MAX)
1857 {
1858 ++m;
1859 ++extra;
1860 }
1861 write(&extra, sizeof(extra));
1862 }
1863 }
1864
1865 write(&state, sizeof(state));
1866 for (int i = 0; i < iNumComponents; i++)
1867 {
1868 if (acComponents[i]->SaveState(f) || ferror(f))
1869 FAILURE(Runtime, "Unable to save component state");

1878bool CSystem::RestoreState(const char* fn)
1879{
1880 std::lock_guard<std::recursive_mutex> bus_lock(device_bus_mutex);
1881 SDeviceAccessScope time_scope(this);
1882 std::unique_ptr<FILE, int (*)(FILE*)> file(fopen(fn, "rb"), &fclose);
1883 if (!file)
1884 {
1885 printf("%%SYS-F-NOFILE: Can't open restore file %s\n", fn);
1886 return false;
1887 }
1888 FILE* f = file.get();
1889 u32 magic = 0, version = 0, system_size = 0, component_count = 0;
1890 u64 memory_size = 0;
1891
1892 if (fread(&magic, sizeof(magic), 1, f) != 1 ||
1893 magic != system_state_magic ||
1894 fread(&version, sizeof(version), 1, f) != 1)
1895 {
1896 printf("%%SYS-F-FORMAT: %s does not appear to be a state file.\n", fn);
1897 return false;
1898 }
1899 if (version != system_state_version)
1900 {
1901 printf("%%SYS-I-VERSION: State file %s is incompatible; "
1902 "version 2.9 is required.\n", fn);
1903 return false;
1904 }
1905 if (fread(&memory_size, sizeof(memory_size), 1, f) != 1 ||
1906 fread(&system_size, sizeof(system_size), 1, f) != 1 ||
1907 fread(&component_count, sizeof(component_count), 1, f) != 1 ||
1908 memory_size != (U64(1) << iNumMemoryBits) ||
1909 system_size != sizeof(state) || component_count != (u32)iNumComponents)
1910 {
1911 printf("%%SYS-F-CONFIG: State file %s has an incomplete or "
1912 "incompatible system header.\n", fn);
1913 return false;
1914 }
1915
1916 // Validate every device and backing medium before replacing any guest RAM.
1917 try
1918 {
1919 for (int i = 0; i < iNumComponents; ++i)
1920 acComponents[i]->prepare_snapshot();
1921 for (int i = 0; i < iNumComponents; ++i)
1922 {
1923 u32 length = 0;
1924 if (fread(&length, sizeof(length), 1, f) != 1 ||
1925 !length || length > snapshot_identity_limit)
1926 {
1927 printf("%%SYS-F-MANIFEST: Invalid state device manifest.\n");
1928 return false;
1929 }
1930 std::string identity(length, '\0');
1931 if (fread(&identity[0], length, 1, f) != 1 ||
1932 identity != acComponents[i]->snapshot_identity())
1933 {
1934 printf("%%SYS-F-MANIFEST: State device or media does not match %s.\n",
1935 acComponents[i]->devid_string);
1936 return false;
1937 }
1938 }
1939 }
1940 catch (const CException& e)
1941 {
1942 printf("%%SYS-F-MANIFEST: Unable to verify state media: %s\n",
1943 e.displayText().c_str());
1944 return false;
1945 }
1946
1947 // After mutation begins a failure must stop emulation, rather than resume.
1948 const auto read = [f](void* data, size_t size)
1949 {
1950 if (fread(data, size, 1, f) != 1)
1951 FAILURE(Runtime, "Incomplete system state");
1952 };
1953 int* mem = (int*)memory;
1954 const u64 memints = memory_size / sizeof(int);
1955 for (u64 m = 0; m < memints;)
1956 {
1957 int value;
1958 read(&value, sizeof(value));
1959 if (value)
1960 mem[m++] = value;
1961 else
1962 {
1963 u32 extra;
1964 read(&extra, sizeof(extra));
1965 if ((u64)extra >= memints - m)
1966 FAILURE(Runtime, "Invalid zero run in system state");
1967 const u64 end = m + (u64)extra + 1;
1968 while (m < end)
1969 mem[m++] = 0;
1970 }
1971 }
1972
1973 read(&state, sizeof(state));
1974
1975 // components
1976 //
1977 // Components should also save any non-initial memory-registrations and re-register upon restore!
1978 //
1979 for (int i = 0; i < iNumComponents; i++)
1980 {
1981 if (acComponents[i]->RestoreState(f) || ferror(f) || feof(f))
1982 FAILURE(Runtime, "Unable to restore system state");
1983 }
1984 for (int i = 0; i < iNumComponents; ++i)
1985 acComponents[i]->finalize_restore();
1986 update_pchip_error_irqs();
1987 return true;
1988}
