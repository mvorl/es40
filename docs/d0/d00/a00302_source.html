165void CCirrus::run()
166{
167 try
168 {
169 // initialize the GUI (and let it know our tilesize)
170 bx_gui->init(state.x_tilesize, state.y_tilesize);
171 for (;;)
172 {
173 // Terminate thread if StopThread is set to true
174 if (StopThread)
175 return;
176 // Handle GUI events (100 times per second)
177 for (int i = 0; i < 10; i++)
178 {
179 bx_gui->lock();
180 bx_gui->handle_events();
181 bx_gui->unlock();
182 CThread::sleep(10);
183 }
184 //Update the screen (10 times per second)
185 bx_gui->lock();
186 update();
187 bx_gui->flush();
188 bx_gui->unlock();
189 }
190 }
191
192 catch (CException& e)
193 {
194 printf("Exception in Cirrus thread: %s.\n", e.displayText().c_str());
195
196 // Let the thread die...
197 }
198}

1012void CCirrus::write_b_3c0(u8 value)
1013{
1014 // Variables to save old state (to detect transitions)
1015 bool prev_video_enabled;
1016 bool prev_line_graphics;
1017 bool prev_int_pal_size;
1018
1019 /* The flip-flop determines whether the write goes to the index-register
1020 (address) or the data-register. */
1021 if (state.attribute_ctrl.flip_flop == 0)
1022 {
1023 // Write goes to the index-register.
1024
1025 /* The index register also has a bit that controls whether video
1026 output is enabled or not.
1027 We check this bit, and compare it to it's previous state, to
1028 determine whether we need to perform an enable or disable
1029 transition. */
1030 prev_video_enabled = state.attribute_ctrl.video_enabled;
1031 state.attribute_ctrl.video_enabled = (value >> 5) & 0x01;
1032#if defined(DEBUG_VGA)
1033 printf("io write 3c0: video_enabled = %u \n",
1034 (unsigned)state.attribute_ctrl.video_enabled);
1035#endif
1036 if (state.attribute_ctrl.video_enabled == 0)
1037 {
1038 if (prev_video_enabled)
1039 {
1040#if defined(DEBUG_VGA)
1041 printf("found disable transition \n");
1042#endif
1043 // Video output has been disabled. Clear the screen.
1044 bx_gui->lock();
1045 bx_gui->clear_screen();
1046 bx_gui->unlock();
1047 }
1048 }
1049 else if (!prev_video_enabled)
1050 {
1051#if defined(DEBUG_VGA)
1052 printf("found enable transition \n");
1053#endif
1054 // Video output has been enabled. Draw the screen.
1055 redraw_area(0, 0, old_iWidth, old_iHeight);
1056 }
1057
1058 // Determine what register should be addressed.
1059 value &= 0x1f; /* address = bits 0..4 */
1060 state.attribute_ctrl.address = value;
1061
1062 /* Registers 0x00..0x0f are palette selection registers.
1063 Write a debugging message for all other registers. */
1064#if defined(DEBUG_VGA)
1065 if (value > 0x0f)
