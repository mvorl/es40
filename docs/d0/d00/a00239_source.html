1/* ES40 emulator -- host-side idle detection methods
2 *
3 * SPDX-License-Identifier: BSD-1-Clause
4 *
5 * Optional JIT heuristics for guests that spin instead of issuing WTINT,
6 * such as HP OpenVMS Alpha 8.4. Disabled by default; repeated PCs alone do
7 * not establish that the guest is idle.
8 *
9 * The configured OS name selects a heuristic, not an OS identity check.
10 * Selecting the wrong name can still trigger sleeps during useful work.
11 * With idle_detect="off" and idle_nap=false (both defaults), the shared
12 * idle wait returns without sleeping. Other timer pacing is independent.
13 *
14 * Prior art: SIMH "SET CPU IDLE=VMS" (MIT, Copyright (c) Robert M Supnik).
15 * No code copied -- only the approach of testing processor state instead of
16 * looking for a particular address.
17 */

23/* Internal rules. Several guest OSes can share one, and the user never names
24 * a rule -- only an OS label, so a rule can be improved without anyone's
25 * config changing. Same split as SIMH's os_tab[].

27enum EIdleRule

29 RULE_NONE = 0,

41/* One row per guest OS, the way SIMH's os_tab[] works. Historical idle host
42 * CPU measurements before interruptible waits, single CPU with the JIT on:
43 *
44 * HP OpenVMS Alpha 8.4 101.2% -> 7.5% vms
45 * Tru64 UNIX 5.1B-4 100.8% -> 9.4% tru64
46 * NetBSD 10.1/alpha stock GENERIC 100.8% -> 20.7% netbsd
47 *
48 * RULE_SPIN cannot distinguish idle loops from repetitive computation and
49 * can repeatedly sleep during useful work. tru64 and netbsd both select it:
50 * reduced host CPU use trades off against guest throughput and latency.
51 * RULE_SPIN_IPL3 additionally samples the VMS PAL PS layout in p22; it does
52 * not prove that an entire dispatch stayed at IPL 3 or did no useful work.
53 * Other CPU/PAL combinations need an appropriate state decoder and explicit
54 * eligibility before relying on this qualifier. These historical figures
55 * do not establish guest clock accuracy or active-workload performance.
56 */
57static const SIdleMethod idle_methods[] =

59 { "off", RULE_NONE, "no host-side idle detection (default)" },
60 { "vms", RULE_SPIN_IPL3, "OpenVMS" },
61 { "tru64", RULE_SPIN, "Tru64 UNIX" },
62 { "netbsd", RULE_SPIN, "NetBSD/alpha" },
63 { 0, 0, 0 }

66inline int idle_method_by_name(const char* n)
67{
68 if (!n)
69 return RULE_NONE;
70 for (int i = 0; idle_methods[i].name; i++)
71 if (!strcasecmp(n, idle_methods[i].name))
72 return idle_methods[i].rule;
73 return -1; // caller reports the bad name rather than silently idling wrong
74}
