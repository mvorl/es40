55static uint32_t radeon_setting(
56 CConfigurator* cfg, const char* key, uint32_t def)
57{
58 const char* text = cfg->get_text_value(key, nullptr);
59 if (!text || !*text)
60 return def;
61 char* end = nullptr;
62 errno = 0;
63 const unsigned long long n = std::strtoull(text, &end, 0);
64 if (errno || *end || text[0] == '-' || n > 0xffffffffULL)
65 throw std::runtime_error(
66 std::string("Invalid radeon integer setting: ") + key);
67 return static_cast<uint32_t>(n);
68}

70static uint32_t radeon_bounded_setting(
71 CConfigurator* cfg, const char* key, uint32_t def, uint32_t max)
72{
73 const auto v = radeon_setting(cfg, key, def);
74 if (v > max)
75 throw std::runtime_error(
76 std::string("radeon setting out of range: ") + key);
77 return v;
78}

91static void radeon_write_u32(FILE* f, uint32_t v)
92{
93 uint8_t b[4];
94 for (unsigned i = 0; i < 4; ++i)
95 b[i] = uint8_t(v >> (8 * i));
96 if (fwrite(b, 1, 4, f) != 4)
97 throw std::runtime_error("Snapshot write failed");
98}

100static uint32_t radeon_read_u32(FILE* f)
101{
102 uint8_t b[4];
103 if (fread(b, 1, 4, f) != 4)
104 throw std::runtime_error("Snapshot read failed");
105 return uint32_t(b[0]) | (uint32_t(b[1]) << 8) | (uint32_t(b[2]) << 16) |
106 (uint32_t(b[3]) << 24);
107}

109static bool radeon_vga_window_offset(u32& address, unsigned selection)
110{
111 switch (selection & 3)
112 {
113 case 1:
114 return address < 0x10000;
115 case 2:
116 if (address < 0x10000 || address >= 0x18000)
117 return false;
118 address -= 0x10000;
119 break;
120 case 3:
121 if (address < 0x18000 || address >= 0x20000)
122 return false;
123 address -= 0x18000;
124 break;
125 default:
126 return address < 0x20000;
127 }
128 return true;
129}

626// VGA memory and display
627
628u8 CRadeon7500::mem_r(offs_t address)
629{
630 if (!radeon_vga_window_offset(address, vga.gc.memory_map_sel))
631 return 0xff;
632 unsigned selected = vga.gc.read_map_sel & 3;
633 if (vga.sequencer.data[4] & 8)
634 {
635 // Keep packed chain-four bytes in the layout consumed by CVGA scanout.
636 for (unsigned plane = 0; plane < 4; ++plane)
637 vga.gc.latch[plane] = vga.memory[(address & ~3u) + plane];
638 selected = address & 3;
639 }
640 else if (!(vga.sequencer.data[4] & 4) && vga.gc.host_oe)
641 {
642 // CVGA text stores character/attribute pairs together; fonts stay planar.
643 vga.gc.latch[0] = vga.memory[address & ~1u];
644 vga.gc.latch[1] = vga.memory[(address & ~1u) + 1];
645 for (unsigned plane = 2; plane < 4; ++plane)
646 vga.gc.latch[plane] = vga.memory[(address >> 1) + plane * 0x10000];
647 selected = (vga.gc.read_map_sel & 2) | (address & 1);
648 }
649 else
650 {
651 for (unsigned plane = 0; plane < 4; ++plane)
652 vga.gc.latch[plane] =
653 vga.memory[(address & 0xffff) + plane * 0x10000];
654 }
655 if (!vga.gc.read_mode)
656 return vga.gc.latch[selected];
657 u8 result = 0xff;
658 for (unsigned plane = 0; plane < 4; ++plane)
659 if (vga.gc.color_dont_care & (1 << plane))
660 result &=
661 u8(vga.gc.latch[plane] ^
662 ((vga.gc.color_compare & (1 << plane)) ? 0 : 0xff));
663 return result;
664}
665
666void CRadeon7500::mem_w(offs_t address, u8 data)
667{
668 if (!radeon_vga_window_offset(address, vga.gc.memory_map_sel))
669 return;
670 if (vga.sequencer.data[4] & 8)
671 {
672 if (vga.sequencer.map_mask & (1 << (address & 3)))
673 vga.memory[address] = vga_latch_write(address & 3, data);
674 return;
675 }
676 if (!(vga.sequencer.data[4] & 4) && vga.gc.host_oe)
677 {
678 const unsigned parity = address & 1;
679 if (vga.sequencer.map_mask & (1 << parity))
680 vga.memory[address] = vga_latch_write(parity, data);
681 if (vga.sequencer.map_mask & (1 << (parity + 2)))
682 vga.memory[(address >> 1) + (parity + 2) * 0x10000] =
683 vga_latch_write(parity + 2, data);
684 return;
685 }
686 for (unsigned plane = 0; plane < 4; ++plane)
687 if (vga.sequencer.map_mask & (1 << plane))
688 vga.memory[(address & 0xffff) + plane * 0x10000] =
689 vga_latch_write(plane, data);
690}

718CRV200::Frame CRadeon7500::render_frame()
719{
720 std::lock_guard<std::recursive_mutex> guard(
721 cSystem->get_device_bus_mutex());
722 if (m_pause.load())
723 return {};
724 // CRTC_EXT_DISP_EN selects the native CRTC over the VGA frontend.
725 if (m_rv200.peek(CRV200::CRTC_GEN_CNTL) & CRV200::CRTC_EXT_DISP_EN)
726 return m_rv200.scanout();
727 if (!m_vga_enable || !vga.crtc.sync_en ||
728 (vga.sequencer.data[0] & 3) != 3 || (vga.sequencer.data[1] & 0x20) ||
729 !(vga.attribute.index & 0x20) || !vga.crtc.maximum_scan_line ||
730 (m_rv200.peek(CRV200::CRTC_EXT_CNTL) & CRV200::DISPLAY_DIS))
731 return {};
732 CRV200::Frame frame;
733 frame.width = (vga.crtc.horz_disp_end + 1) *
734 ((vga.gc.alpha_dis || (vga.sequencer.data[1] & 1)) ? 8 : 9);
735 frame.height = vga.crtc.vert_disp_end + 1;
736 if (!frame.width || frame.width > 4096 || frame.height > 1024)
737 return {};
738 // CVGA can write repeated scanlines past the visible-area bottom. Cover CGA
739 // double-scan plus one character row; only the visible rectangle is used.
740 m_render_bitmap.resize(frame.width, frame.height * 2 + 64);
741 screen().set_visible_area(frame.width, frame.height);
742 screen().tick_frame();
743 vga.crtc.start_addr = latch_start_addr();
744 vga.attribute.pel_shift = vga.attribute.pel_shift_latch;
745 palette_update();
746 CVGA::screen_update(
747 m_render_bitmap, rectangle(0, frame.width - 1, 0, frame.height - 1));
748 frame.argb.assign(
749 m_render_bitmap.raw(),
750 m_render_bitmap.raw() + size_t(frame.width) * frame.height);
751 return frame;
752}

823{

849CRadeon7500::OptionROM CRadeon7500::OptionROM::parse(
850 const std::vector<uint8_t>& raw, const std::string& layout)
851{
852 auto fail = [](const char* message) {
853 throw std::invalid_argument(message);
854 };
855 const size_t n = raw.size();
856 if (n < 2048 || n > 1024 * 1024)
857 fail("ROM must be 2 KiB to 1 MiB");
858 if (layout != "exact" && layout != "pad_ff")
859 fail("rom_layout must be exact or pad_ff");
860 if (layout == "exact" && (n & (n - 1)))
861 fail("Non-power-of-two ROM image requires rom_layout=pad_ff");
862 auto word = [&](size_t p) -> uint16_t {
863 if (p > n - 2)
864 fail("ROM field out of bounds");
865 return uint16_t(uint16_t(raw[p]) | (uint16_t(raw[p + 1]) << 8));
866 };
867 size_t at = 0;
868 bool last = false;
869 for (unsigned images = 0; images < 32 && !last; ++images)
870 {
871 if (at + 0x1a > n || raw[at] != 0x55 || raw[at + 1] != 0xaa)
872 fail("ROM missing 55 AA signature");
873 const size_t p = at + word(at + 0x18);
874 if (p < at + 0x1a || p + 0x18 > n || raw[p] != 'P' ||
875 raw[p + 1] != 'C' || raw[p + 2] != 'I' || raw[p + 3] != 'R')
876 fail("ROM missing/bounded PCIR structure");
877 if (word(p + 4) != 0x1002 || word(p + 6) != 0x5157)
878 fail("ROM PCIR identity is not ATI 1002:5157 (RV200)");
879 const size_t length = size_t(word(p + 0x10)) * 512;
880 if (!length || length > n - at || p + 0x18 > at + length)
881 fail("Invalid option-ROM image length");
882 const size_t pcir_length = word(p + 0x0a);
883 if (pcir_length < 0x18 || pcir_length > at + length - p)
884 fail("Invalid PCIR structure length");
885 if (raw[p + 0x14] == 0 &&
886 (!raw[at + 2] || size_t(raw[at + 2]) * 512 > length))
887 fail("Invalid x86 initialization image length");
888 unsigned sum = 0;
889 for (size_t i = at; i < at + length; ++i)
890 sum += raw[i];
891 if (sum & 255)
892 fail("Option-ROM checksum failed");
893 if (raw[p + 0x14] != 0 && raw[p + 0x14] != 3)
894 fail("Only x86 and EFI option-ROM images are supported");
895 last = (raw[p + 0x15] & 0x80) != 0;
896 at += length;
897 }
898 if (!last)
899 fail("Unterminated option-ROM image chain");
900 OptionROM rom;
901 rom.file_size = static_cast<uint32_t>(n);
902 const std::string agp = "V200AGP";
903 if (std::search(raw.begin(), raw.end(), agp.begin(), agp.end()) !=
904 raw.end())
905 fail("AGP ROM rejected for PCI 3X-PBXGG-AA target; choose a PCI image");
906 rom.bytes = raw;
907 if (layout == "pad_ff")
908 {
909 // Erased padding to the next power of two; the upper decode is unmeasured.
910 size_t decoded = 2048;
911 while (decoded < n)
912 decoded *= 2;
913 rom.bytes.resize(decoded, 0xff);
914 }
915 return rom;
916}

1179bool CRadeon7500::decodes_memory_access(
1180 int index, u64 address, int dsize, bool write) const noexcept
1181{
1182 if (index >= PCI_RANGE_BASE)
1183 return CPCIDevice::decodes_memory_access(index, address, dsize, write);
1184 if (dsize != 8 && dsize != 16 && dsize != 32 && dsize != 64)
1185 return false;
1186 const unsigned bytes = unsigned(dsize) / 8;
1187 if (index == LegacyIO)
1188 {
1189 if (!legacy_enabled(false) || address >= 0x30 || bytes > 0x30 - address)
1190 return false;
1191 // Claim an access only when every byte belongs to this device; reserved
1192 // port bytes must not make us a competing positive decoder.
1193 for (unsigned i = 0; i < bytes; ++i)
1194 if (!vga_port_enabled(0x3b0 + u32(address) + i))
1195 return false;
1196 return true;
1197 }
1198 if (index != LegacyMemory || !legacy_enabled(true) || address >= 0x20000 ||
1199 bytes > 0x20000 - address)
1200 return false;
1201 switch (vga.gc.memory_map_sel & 3)
1202 {
1203 case 1:
1204 return address + bytes <= 0x10000;
1205 case 2:
1206 return address >= 0x10000 && address + bytes <= 0x18000;
1207 case 3:
1208 return address >= 0x18000;
1209 default:
1210 return true;
1211 }
1212}

1214u32 CRadeon7500::ReadMem_Legacy(int index, u32 address, int dsize)
1215{
1216 std::lock_guard<std::recursive_mutex> guard(
1217 cSystem->get_device_bus_mutex());
1218 if (!decodes_memory_access(index, address, dsize, false))
1219 return 0xffffffff;
1220 if (dsize != 8 && dsize != 16 && dsize != 32)
1221 return 0xffffffff;
1222 tick();
1223 u32 data = 0;
1224 for (int i = 0; i < dsize / 8; ++i)
1225 data |= u32(index == LegacyIO ? io_read_b(0x3b0 + address + i)
1226 : mem_r(address + i))
1227 << (i * 8);
1228 if (m_trace)
1229 m_trace << "# VGA R " << index << ' ' << std::hex << address << ' '
1230 << data << std::dec << " bits=" << dsize << '\n';

1449{

1452 bytes.push_back(m_crtc_map.read_byte(i));
1453 for (unsigned i = 0; i < 5; ++i)
1454 bytes.push_back(m_seq_map.read_byte(i));
1455 for (unsigned i = 0; i < 9; ++i)
1456 bytes.push_back(m_gc_map.read_byte(i));
1457 for (unsigned i = 0; i < 21; ++i)
1458 bytes.push_back(m_atc_map.read_byte(i));
1459 bytes.insert(
1460 bytes.end(),
1461 {vga.crtc.index,
1462 vga.sequencer.index,
1463 vga.gc.index,
1464 vga.attribute.index,

1466 vga.feature_control,
1467 m_vga_enable,
1468 vga.dac.mask,
1469 u8(vga.attribute.state),
1470 u8(vga.dac.read),
1471 u8(vga.dac.state)});
1472 for (const u8 value : vga.dac.loading)
1473 bytes.push_back(value);
1474 for (const u8 value : vga.gc.latch)
1475 bytes.push_back(value);
1476 for (unsigned i = 0; i < 4; ++i)

1488void CRadeon7500::restore_vga_state(const std::vector<u8>& bytes)
1489{
1490 auto dword = [&](size_t offset) {
1491 return u32(bytes[offset]) | (u32(bytes[offset + 1]) << 8) |
1492 (u32(bytes[offset + 2]) << 16) | (u32(bytes[offset + 3]) << 24);
1493 };
1494 if (bytes.size() != VGAStateSize || bytes[0] != 1 || bytes[67] > 1 ||
1495 bytes[69] > 1 || bytes[70] > 1 || bytes[71] > 2 || dword(79) > 0xffff ||
1496 bytes[83] > 15 || bytes[84] > 1 || bytes[85] > 1 || dword(86) > 1024 ||
1497 dword(90) > 31)
1498 throw std::runtime_error("Invalid VGA snapshot fields");
1499 reset_vga();
1500 // Restore CR11 protection LAST so CR0-7 are not discarded while loading.
1501 for (unsigned i = 0; i < 25; ++i)
1502 if (i != 0x11)
1503 m_crtc_map.write_byte(i, bytes[1 + i]);
1504 m_crtc_map.write_byte(0x11, bytes[1 + 0x11]);
1505 for (unsigned i = 0; i < 5; ++i)
1506 m_seq_map.write_byte(i, bytes[26 + i]);
1507 for (unsigned i = 0; i < 9; ++i)
1508 m_gc_map.write_byte(i, bytes[31 + i]);
1509 for (unsigned i = 0; i < 21; ++i)
1510 m_atc_map.write_byte(i, bytes[40 + i]);
1511 // Every register must read back exactly as saved.
1512 for (unsigned i = 0; i < 25; ++i)
1513 if (m_crtc_map.read_byte(i) != bytes[1 + i])
1514 throw std::runtime_error("Invalid CRTC snapshot encoding");
1515 for (unsigned i = 0; i < 5; ++i)
1516 if (m_seq_map.read_byte(i) != bytes[26 + i])
1517 throw std::runtime_error("Invalid sequencer snapshot encoding");
1518 for (unsigned i = 0; i < 9; ++i)
1519 if (m_gc_map.read_byte(i) != bytes[31 + i])
1520 throw std::runtime_error(
1521 "Invalid graphics-controller snapshot encoding");
1522 for (unsigned i = 0; i < 21; ++i)
1523 if (m_atc_map.read_byte(i) != bytes[40 + i])
1524 throw std::runtime_error("Invalid attribute snapshot encoding");
1525 vga.crtc.index = bytes[61];
1526 vga.sequencer.index = bytes[62];
1527 vga.gc.index = bytes[63];
1528 vga.attribute.index = bytes[64];
1529 vga.miscellaneous_output = bytes[65];
1530 m_ioas = (bytes[65] & 1) != 0;
1531 vga.feature_control = bytes[66];
1532 m_vga_enable = bytes[67];
1533 vga.dac.mask = bytes[68];
1534 vga.attribute.state = bytes[69];
1535 vga.dac.read = bytes[70];
1536 vga.dac.state = bytes[71];
1537 for (unsigned i = 0; i < 3; ++i)
1538 vga.dac.loading[i] = bytes[72 + i];
1539 for (unsigned i = 0; i < 4; ++i)
1540 vga.gc.latch[i] = bytes[75 + i];
1541 vga.crtc.start_addr = dword(79);
1542 vga.attribute.pel_shift = bytes[83];
1543 vga.cursor.visible = bytes[84];
1544 vga.crtc.irq_latch = bytes[85];
1545 m_vga_scanline = dword(86);
1546 m_vga_status_phase = dword(90);
1547 vga.dac.dirty = 1;
1548}

1585int CRadeon7500::RestoreState(FILE* f)
1586{
1587 if (!f)
1588 return -1;
1589 std::lock_guard<std::recursive_mutex> guard(
1590 cSystem->get_device_bus_mutex());
1591 try
1592 {
1593 // PCI state is restored first, following the other ES40 devices.
1594 // A failure aborts the whole system restore; it must not resume the VM.
1595 struct ReplayGuard
1596 {
1597 bool& flag;
1598
1599 ReplayGuard(bool& value) : flag(value) { flag = true; }
1600
1601 ~ReplayGuard() { flag = false; }
1602 } replay(m_replaying_pci);
1603
1604 if (CPCIDevice::RestoreState(f))
1605 return -1;
1606 if (radeon_read_u32(f) != 0x31563752 ||
1607 radeon_read_u32(f) != VGAStateSize)
1608 return -1;
1609 std::vector<u8> vga_bytes(VGAStateSize);
1610 if (fread(vga_bytes.data(), 1, vga_bytes.size(), f) != vga_bytes.size())
1611 return -1;
1612 const u32 lo = radeon_read_u32(f), hi = radeon_read_u32(f),
1613 n = radeon_read_u32(f);
1614 const u32 vram = m_rv200.options().vram_bytes;
1615 if (n < vram || n > vram + 65536 + 16)
1616 return -1;
1617 std::string bytes(n, '\0');
1618 if (fread(bytes.data(), 1, n, f) != n)
1619 return -1;
1620 std::istringstream stream(bytes, std::ios::binary);
1621 m_rv200.RestoreState(stream);
1622 // The core's transactional loader owns its VRAM; refresh our borrowed view.
1623 vga.memory = m_rv200.vram_data();
1624 if (stream.peek() != std::char_traits<char>::eof())
1625 return -1;
1626 restore_vga_state(vga_bytes);
1627 m_access_count = u64(lo) | (u64(hi) << 32);
1628 sync_apertures();
1629 // IRQ output is committed by finalize_restore after all devices restore.
1630 return 0;
1631 }
1632 catch (const std::exception& e)
1633 {
1634 vga.memory = m_rv200.vram_data();
1635 printf(
1636 "%s: Could not restore graphics state: %s.\n",
1637 devid_string,
1638 e.what());
1639 return -1;
1640 }

1651u8 CRadeon7500::input_status()
1652{
1653 vga.attribute.state = 0;
1654 // Deterministic compatibility timing: 32 status reads per synthetic
1655 // scanline. Decode probes and host frame delivery do not clock it.
1656 const bool native =
1657 (m_rv200.peek(CRV200::CRTC_GEN_CNTL) & CRV200::CRTC_EXT_DISP_EN) != 0;
1658 if (++m_vga_status_phase == 32)
1659 {
1660 m_vga_status_phase = 0;
1661 if (native)
1662 m_rv200.advance_scanlines(1);
1663 else
1664 advance_vga_scanlines(1);
1665 }
1666 if (native)
1667 {
1668 // Approximate retrace with the native vertical blank.
1669 const u32 visible =
1670 ((m_rv200.peek(CRV200::CRTC_V_TOTAL_DISP) >> 16) & 0xfff) + 1;
1671 const bool blank = m_rv200.scanline() >= visible;
1672 return u8((blank ? 9 : 0) | (m_vga_status_phase >= 24 ? 1 : 0));
1673 }
1674 const u32 total = u32(vga.crtc.vert_total) + 2;
1675 const u32 start = vga.crtc.vert_retrace_start % total;
1676 u32 length = (vga.crtc.vert_retrace_end - vga.crtc.vert_retrace_start) & 15;
1677 if (!length)
1678 length = 16;
1679 const bool retrace = ((m_vga_scanline + total - start) % total) < length;
1680 const bool blank =
1681 m_vga_scanline > vga.crtc.vert_disp_end || m_vga_status_phase >= 24;
1682 return u8((retrace ? 8 : 0) | (blank ? 1 : 0));
1683}

1685u8 CRadeon7500::io_read_b(u32 port)
1686{
1687 // MMIO aliases bypass the PCI I/O enable, but not the VGA enable or the
1688 // monochrome/colour port selection.
1689 if (!m_initialized || !vga_port_enabled(port))
1690 return 0xff;
1691 switch (port)
1692 {
1693 case 0x3b4:
1694 case 0x3d4:
1695 return crtc_address_r(0);
1696 case 0x3b5:
1697 case 0x3d5:
1698 return crtc_data_r(0);
1699 case 0x3ba:
1700 case 0x3da:
1701 return input_status();
1702 case 0x3c0:
1703 return atc_address_r(0);
1704 case 0x3c1:
1705 return atc_data_r(0);
1706 case 0x3c2:
1707 return input_status_0_r(0);
1708 case 0x3c3:
1709 return m_vga_enable;
1710 case 0x3c4:
1711 return sequencer_address_r(0);
1712 case 0x3c5:
1713 return sequencer_data_r(0);
1714 case 0x3c6:
1715 return vga.dac.mask;
1716 case 0x3c7:
1717 return vga.dac.read ? 3 : 0;
1718 case 0x3c8:
1719 return u8(m_rv200.peek(CRV200::PALETTE_INDEX));
1720 case 0x3c9:
1721 {
1722 // PALETTE_INDEX holds the write index in bits 7:0, read index in 23:16.
1723 const u32 index = (m_rv200.peek(CRV200::PALETTE_INDEX) >> 16) & 255;
1724 u8 value = u8(m_rv200.palette_color(index) >> (16 - 8 * vga.dac.state));
1725 if (!(m_rv200.peek(CRV200::DAC_CNTL) & CRV200::DAC_8BIT_EN))
1726 value >>= 2;
1727 if (++vga.dac.state == 3)
1728 {
1729 vga.dac.state = 0;
1730 m_rv200.WriteMem(
1731 CRV200::PALETTE_INDEX,
1732 32,
1733 (m_rv200.peek(CRV200::PALETTE_INDEX) & ~0xff0000u) |
1734 (((index + 1) & 255) << 16));
1735 }
1736 return value;
1737 }
1738 case 0x3ca:
1739 return feature_control_r(0);
1740 case 0x3cc:
1741 return miscellaneous_output_r(0);
1742 case 0x3ce:
1743 return gc_address_r(0);
1744 case 0x3cf:
1745 return gc_data_r(0);
1746 default:
1747 report_unimplemented("VGA port", port, 0, false);
1748 return 0xff;
1749 }
1750}

1755 return;

1757 {
1758 case 0x3b4:
1759 case 0x3d4:
1760 crtc_address_w(0, data);
1761 break;
1762 case 0x3b5:
1763 case 0x3d5:
1764 crtc_data_w(0, data);
1765 break;
1766 case 0x3ba:
1767 case 0x3da:
1768 feature_control_w(0, data);
1769 break;
1770 case 0x3c0:
1771 atc_address_data_w(0, data);
1772 break;
1773 case 0x3c2:
1774 miscellaneous_output_w(0, data);
1775 break;
1776 case 0x3c3:
1777 m_vga_enable = data & 1;
1778 break;
1779 case 0x3c4:
1780 sequencer_address_w(0, data);
1781 break;
1782 case 0x3c5:
1783 sequencer_data_w(0, data);
1784 break;
1785 case 0x3c6:
1786 vga.dac.mask = data;
1787 vga.dac.dirty = 1;
1788 break;
1789 case 0x3c7:
1790 vga.dac.state = 0;
1791 vga.dac.read = 1;
1792 m_rv200.WriteMem(
1793 CRV200::PALETTE_INDEX,
1794 32,
1795 (m_rv200.peek(CRV200::PALETTE_INDEX) & ~0xff0000u) |

1797 break;
1798 case 0x3c8:
1799 vga.dac.state = 0;
1800 vga.dac.read = 0;
1801 m_rv200.WriteMem(
1802 CRV200::PALETTE_INDEX,
1803 32,
1804 (m_rv200.peek(CRV200::PALETTE_INDEX) & ~255u) | data);
1805 break;
1806 case 0x3c9:
1807 {
1808 const u32 index = m_rv200.peek(CRV200::PALETTE_INDEX) & 255;
1809 // DAC_8BIT_EN clear: expand six-bit VGA components.
1810 if (!(m_rv200.peek(CRV200::DAC_CNTL) & CRV200::DAC_8BIT_EN))
1811 data = u8(((data & 63) << 2) | ((data & 63) >> 4));
1812 vga.dac.loading[vga.dac.state++] = data;
1813 if (vga.dac.state == 3)
1814 {
1815 vga.dac.state = 0;
1816 m_rv200.set_palette_color(
