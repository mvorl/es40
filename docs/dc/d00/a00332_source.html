57static uint32_t powerstorm_setting(
58 CConfigurator* cfg, const char* key, uint32_t def,
59 uint32_t max = 0xffffffff)
60{
61 const char* text = cfg->get_text_value(key, nullptr);
62 if (!text || !*text)
63 return def;
64 char* end = nullptr;
65 errno = 0;
66 const unsigned long long n = std::strtoull(text, &end, 0);
67 if (errno || end == text || *end || text[0] == '-' || n > max)
68 throw std::runtime_error(
69 std::string("Invalid powerstorm setting: ") + key);
70 return static_cast<uint32_t>(n);
71}

73static void powerstorm_write_u32(FILE* f, uint32_t v)
74{
75 uint8_t b[4];
76 for (unsigned i = 0; i < 4; ++i)
77 b[i] = uint8_t(v >> (8 * i));
78 if (fwrite(b, 1, 4, f) != 4)
79 throw std::runtime_error("Snapshot write failed");
80}

82static uint32_t powerstorm_read_u32(FILE* f)
83{
84 uint8_t b[4];
85 if (fread(b, 1, 4, f) != 4)
86 throw std::runtime_error("Snapshot read failed");
87 return uint32_t(b[0]) | (uint32_t(b[1]) << 8) | (uint32_t(b[2]) << 16) |
88 (uint32_t(b[3]) << 24);
89}

91static bool powerstorm_vga_window_offset(u32& address, unsigned selection)
92{
93 switch (selection & 3)
94 {
95 case 1:
96 return address < 0x10000;
97 case 2:
98 if (address < 0x10000 || address >= 0x18000)
99 return false;
100 address -= 0x10000;
101 break;
102 case 3:
103 if (address < 0x18000 || address >= 0x20000)
104 return false;
105 address -= 0x18000;
106 break;
107 default:
108 return address < 0x20000;
109 }
110 return true;
111}

624// VGA memory and display
625
626u8 CPowerStorm3xx::mem_r(offs_t address)
627{
628 if (!powerstorm_vga_window_offset(address, vga.gc.memory_map_sel))
629 return 0xff;
630 unsigned selected = vga.gc.read_map_sel & 3;
631 if (vga.sequencer.data[4] & 8)
632 {
633 // Keep packed chain-four bytes in the layout consumed by CVGA scanout.
634 for (unsigned plane = 0; plane < 4; ++plane)
635 vga.gc.latch[plane] = vga.memory[(address & ~3u) + plane];
636 selected = address & 3;
637 }
638 else if (!(vga.sequencer.data[4] & 4) && vga.gc.host_oe)
639 {
640 // CVGA text stores character/attribute pairs together; fonts stay planar.
641 vga.gc.latch[0] = vga.memory[address & ~1u];
642 vga.gc.latch[1] = vga.memory[(address & ~1u) + 1];
643 for (unsigned plane = 2; plane < 4; ++plane)
644 vga.gc.latch[plane] = vga.memory[(address >> 1) + plane * 0x10000];
645 selected = (vga.gc.read_map_sel & 2) | (address & 1);
646 }
647 else
648 {
649 for (unsigned plane = 0; plane < 4; ++plane)
650 vga.gc.latch[plane] =
651 vga.memory[(address & 0xffff) + plane * 0x10000];
652 }
653 if (!vga.gc.read_mode)
654 return vga.gc.latch[selected];
655 u8 result = 0xff;
656 for (unsigned plane = 0; plane < 4; ++plane)
657 if (vga.gc.color_dont_care & (1 << plane))
658 result &=
659 u8(vga.gc.latch[plane] ^
660 ((vga.gc.color_compare & (1 << plane)) ? 0 : 0xff));
661 return result;
662}
663
664void CPowerStorm3xx::mem_w(offs_t address, u8 data)
665{
666 if (!powerstorm_vga_window_offset(address, vga.gc.memory_map_sel))
667 return;
668 if (vga.sequencer.data[4] & 8)
669 {
670 if (vga.sequencer.map_mask & (1 << (address & 3)))
671 vga.memory[address] = vga_latch_write(address & 3, data);
672 return;
673 }
674 if (!(vga.sequencer.data[4] & 4) && vga.gc.host_oe)
675 {
676 const unsigned parity = address & 1;
677 if (vga.sequencer.map_mask & (1 << parity))
678 vga.memory[address] = vga_latch_write(parity, data);
679 if (vga.sequencer.map_mask & (1 << (parity + 2)))
680 vga.memory[(address >> 1) + (parity + 2) * 0x10000] =
681 vga_latch_write(parity + 2, data);
682 return;
683 }
684 for (unsigned plane = 0; plane < 4; ++plane)
685 if (vga.sequencer.map_mask & (1 << plane))
686 vga.memory[(address & 0xffff) + plane * 0x10000] =
687 vga_latch_write(plane, data);
688}

830
831std::array<u32, 64> CPowerStorm3xx::PCIConfig::config_data() const
832{
833 std::array<u32, 64> data{};
834 data[0] = u32(vendor) | (u32(device) << 16);
835 data[2] = (class_code << 8) | revision;
836 data[6] = 0x00000001; // BAR2: native index/data I/O pair
837 data[11] = u32(subsystem_vendor) | (u32(subsystem_device) << 16);
838 data[15] = 0x000001ff; // INTA#, interrupt line unassigned

841
842std::array<u32, 64> CPowerStorm3xx::PCIConfig::config_mask(u32 rom_size) const
843{
844 std::array<u32, 64> mask{};
845 mask[1] = 0x00000147; // I/O, memory, bus master, parity response, SERR
846 mask[3] = 0x0000ffff; // cache-line size and latency timer
847 mask[4] = ~(bar0_size - 1);
848 mask[5] = ~(bar1_size - 1);
849 mask[6] = ~(IOBARSize - 1);
850 if (rom_size)
851 mask[12] = (~(rom_size - 1) & 0xfffff800u) | 1; // ROM decode and enable
852 mask[15] = 0xff; // interrupt line only; pin fixed
853 return mask;
854}
855
856CPowerStorm3xx::OptionROM CPowerStorm3xx::OptionROM::parse(
857 const std::vector<u8>& raw, const std::string& layout)
858{
859 auto fail = [](const char* message) {
860 throw std::invalid_argument(message);
861 };
862 const size_t n = raw.size();
863 if (n < 2048 || n > 1024 * 1024)
864 fail("ROM must be 2 KiB to 1 MiB");
865 if (layout != "exact" && layout != "pad_ff")
866 fail("rom_layout must be exact or pad_ff");
867 if (layout == "exact" && (n & (n - 1)))
868 fail("Non-power-of-two ROM image requires rom_layout=pad_ff");
869 auto word = [&](size_t p) -> u16 {
870 if (p > n - 2)
871 fail("ROM field out of bounds");
872 return u16(raw[p] | (u16(raw[p + 1]) << 8));
873 };
874 size_t at = 0;
875 bool last = false;
876 OptionROM rom;
877 for (unsigned images = 0; images < 32 && !last; ++images)
878 {
879 if (at + 0x1a > n || raw[at] != 0x55 || raw[at + 1] != 0xaa)
880 fail("ROM missing 55 AA signature");
881 const size_t p = at + word(at + 0x18);
882 if (p < at + 0x1a || p + 0x18 > n || raw[p] != 'P' ||
883 raw[p + 1] != 'C' || raw[p + 2] != 'I' || raw[p + 3] != 'R')
884 fail("ROM missing/bounded PCIR structure");
885 if (word(p + 4) != 0x10ba || word(p + 6) != 0x0304)
886 fail("ROM PCIR identity is not 10BA:0304 (PowerStorm 300/350)");
887 const size_t length = size_t(word(p + 0x10)) * 512;
888 if (!length || length > n - at || p + 0x18 > at + length)
889 fail("Invalid option-ROM image length");
890 if (word(p + 0x0a) < 0x18 || word(p + 0x0a) > at + length - p)
891 fail("Invalid PCIR structure length");
892 if (raw[p + 0x14] != 0)
893 fail("Only an x86 option-ROM image is supported");
894 if (!raw[at + 2] || size_t(raw[at + 2]) * 512 > length)
895 fail("Invalid x86 initialization image length");
896 unsigned sum = 0;
897 for (size_t i = at; i < at + length; ++i)
898 sum += raw[i];
899 if (sum & 255)
900 fail("Option-ROM checksum failed");
901 rom.pcir_class = u32(raw[p + 13]) | (u32(raw[p + 14]) << 8) |
902 (u32(raw[p + 15]) << 16);
903 last = (raw[p + 0x15] & 0x80) != 0;
904 at += length;
905 }
906 if (!last)
907 fail("Unterminated option-ROM image chain");
908 // Keep the supplied image unmodified; only an erased tail may follow it.
909 for (size_t i = at; i < n; ++i)
910 if (raw[i] != 0xff)
911 fail("Unexpected data after last option-ROM image");
912 rom.bytes = raw;
913 rom.file_size = u32(n);
914 rom.image_bytes = u32(at);
915 if (layout == "pad_ff")
916 {
917 size_t decoded = 2048;
918 while (decoded < n)
919 decoded *= 2;
920 rom.bytes.resize(decoded, 0xff);
921 }
922 return rom;
923}

1181
1182bool CPowerStorm3xx::decodes_memory_access(
1183 int index, u64 address, int dsize, bool write) const noexcept
1184{
1185 if (index >= PCI_RANGE_BASE)
1186 return CPCIDevice::decodes_memory_access(index, address, dsize, write);
1187 if (dsize != 8 && dsize != 16 && dsize != 32 && dsize != 64)
1188 return false;
1189 const unsigned bytes = unsigned(dsize) / 8;
1190 if (index == LegacyIO)
1191 {
1192 if (!legacy_enabled(false) || address >= 0x30 || bytes > 0x30 - address)
1193 return false;
1194 // Claim only accesses whose every byte decodes to this device.
1195 for (unsigned i = 0; i < bytes; ++i)
1196 if (!vga_port_enabled(0x3b0 + u32(address) + i))
1197 return false;
1198 return true;
1199 }
1200 if (index != LegacyMemory || !legacy_enabled(true) || address >= 0x20000 ||
1201 bytes > 0x20000 - address)
1202 return false;
1203 switch (vga.gc.memory_map_sel & 3)
1204 {
1205 case 1:
1206 return address + bytes <= 0x10000;
1207 case 2:
1208 return address >= 0x10000 && address + bytes <= 0x18000;
1209 case 3:
1210 return address >= 0x18000;
1211 default:
1212 return true;
1213 }
1214}

1216u32 CPowerStorm3xx::ReadMem_Legacy(int index, u32 address, int dsize)
1217{
1218 std::lock_guard<std::recursive_mutex> guard(
1219 cSystem->get_device_bus_mutex());
1220 if (!decodes_memory_access(index, address, dsize, false))

1308{

1316 return false;

1318 }

1510{

1512 for (unsigned i = 0; i < 25; ++i)
1513 bytes.push_back(m_crtc_map.read_byte(i));
1514 for (unsigned i = 0; i < 5; ++i)
1515 bytes.push_back(m_seq_map.read_byte(i));
1516 for (unsigned i = 0; i < 9; ++i)
1517 bytes.push_back(m_gc_map.read_byte(i));
1518 for (unsigned i = 0; i < 21; ++i)
1519 bytes.push_back(m_atc_map.read_byte(i));
1520 bytes.insert(
1521 bytes.end(),
1522 {vga.crtc.index,
1523 vga.sequencer.index,
1524 vga.gc.index,
1525 vga.attribute.index,
1526 vga.miscellaneous_output,
1527 vga.feature_control,
1528 m_vga_enable,
1529 vga.dac.mask,
1530 u8(vga.attribute.state),
1531 u8(vga.dac.read),
1532 u8(vga.dac.state)});
1533 for (const u8 value : vga.dac.loading)
1534 bytes.push_back(value);
1535 for (const u8 value : vga.gc.latch)
1536 bytes.push_back(value);
1537 for (unsigned i = 0; i < 4; ++i)
1538 bytes.push_back(u8(vga.crtc.start_addr >> (8 * i)));
1539 bytes.push_back(vga.attribute.pel_shift);
1540 bytes.push_back(vga.cursor.visible);
1541 bytes.push_back(vga.crtc.irq_latch);
1542 for (unsigned i = 0; i < 4; ++i)
1543 bytes.push_back(u8(m_vga_scanline >> (8 * i)));
1544 for (unsigned i = 0; i < 4; ++i)
1545 bytes.push_back(u8(m_vga_status_phase >> (8 * i)));
1546 bytes.push_back(u8(m_crtc_io_base));
1547 bytes.push_back(u8(m_crtc_io_base >> 8));
1548 return bytes;
1549}
1550
1551void CPowerStorm3xx::restore_vga_state(const std::vector<u8>& bytes)
1552{
1553 auto dword = [&](size_t offset) {
1554 return u32(bytes[offset]) | (u32(bytes[offset + 1]) << 8) |
1555 (u32(bytes[offset + 2]) << 16) | (u32(bytes[offset + 3]) << 24);
1556 };
1557 if (bytes.size() != VGAStateSize || bytes[0] != 1 || bytes[67] > 1 ||
1558 bytes[69] > 1 || bytes[70] > 1 || bytes[71] > 2 || dword(79) > 0xffff ||
1559 bytes[83] > 15 || bytes[84] > 1 || bytes[85] > 1 || dword(86) > 1024 ||
1560 dword(90) > 31)
1561 throw std::runtime_error("Invalid VGA snapshot fields");
1562 reset_vga();
1563 // Restore CR11 protection LAST so CR0-7 are not discarded while loading.
1564 for (unsigned i = 0; i < 25; ++i)
1565 if (i != 0x11)
1566 m_crtc_map.write_byte(i, bytes[1 + i]);
1567 m_crtc_map.write_byte(0x11, bytes[1 + 0x11]);
1568 for (unsigned i = 0; i < 5; ++i)
1569 m_seq_map.write_byte(i, bytes[26 + i]);
1570 for (unsigned i = 0; i < 9; ++i)
1571 m_gc_map.write_byte(i, bytes[31 + i]);
1572 for (unsigned i = 0; i < 21; ++i)
1573 m_atc_map.write_byte(i, bytes[40 + i]);
1574 // Every register must read back exactly as saved.
1575 for (unsigned i = 0; i < 25; ++i)
1576 if (m_crtc_map.read_byte(i) != bytes[1 + i])
1577 throw std::runtime_error("Invalid CRTC snapshot encoding");
1578 for (unsigned i = 0; i < 5; ++i)
1579 if (m_seq_map.read_byte(i) != bytes[26 + i])
1580 throw std::runtime_error("Invalid sequencer snapshot encoding");
1581 for (unsigned i = 0; i < 9; ++i)
1582 if (m_gc_map.read_byte(i) != bytes[31 + i])
1583 throw std::runtime_error(
1584 "Invalid graphics-controller snapshot encoding");
1585 for (unsigned i = 0; i < 21; ++i)
1586 if (m_atc_map.read_byte(i) != bytes[40 + i])
1587 throw std::runtime_error("Invalid attribute snapshot encoding");
1588 vga.crtc.index = bytes[61];
1589 vga.sequencer.index = bytes[62];
1590 vga.gc.index = bytes[63];
1591 vga.attribute.index = bytes[64];
1592 vga.miscellaneous_output = bytes[65];
1593 m_ioas = (bytes[65] & 1) != 0;
1594 vga.feature_control = bytes[66];
1595 m_vga_enable = bytes[67];
1596 vga.dac.mask = bytes[68];
1597 vga.attribute.state = bytes[69];
1598 vga.dac.read = bytes[70];
1599 vga.dac.state = bytes[71];
1600 for (unsigned i = 0; i < 3; ++i)
1601 vga.dac.loading[i] = bytes[72 + i];
1602 for (unsigned i = 0; i < 4; ++i)
1603 vga.gc.latch[i] = bytes[75 + i];
1604 vga.crtc.start_addr = dword(79);
1605 vga.attribute.pel_shift = bytes[83];
1606 vga.cursor.visible = bytes[84];
1607 vga.crtc.irq_latch = bytes[85];
1608 m_vga_scanline = dword(86);
1609 m_vga_status_phase = dword(90);
1610 m_crtc_io_base = u16(bytes[94] | (u16(bytes[95]) << 8));
1611 vga.dac.dirty = 1;
1612}

1654
1655int CPowerStorm3xx::RestoreState(FILE* f)
1656{
1657 if (!f)
1658 return -1;
1659 std::lock_guard<std::recursive_mutex> guard(
1660 cSystem->get_device_bus_mutex());
1661 try
1662 {
1663 // Restore PCI first; any restore failure must abort the VM restore.
1664 struct ReplayGuard
1665 {
1666 bool& flag;
1667
1668 ReplayGuard(bool& value) : flag(value) { flag = true; }
1669
1670 ~ReplayGuard() { flag = false; }
1671 } replay(m_replaying_pci);
1672
1673 if (CPCIDevice::RestoreState(f) || powerstorm_read_u32(f) != 0x31475350)
1674 return -1;
1675 const u32 identity_size = powerstorm_read_u32(f);
1676 if (identity_size > 8192)
1677 return -1;
1678 std::string identity(identity_size, '\0');
1679 if (identity_size &&
1680 fread(&identity[0], 1, identity_size, f) != identity_size)
1681 return -1;
1682 if (identity != snapshot_identity() ||
1683 powerstorm_read_u32(f) != VGAStateSize)
1684 return -1;
1685 std::vector<u8> vga_bytes(VGAStateSize);
1686 if (fread(vga_bytes.data(), 1, vga_bytes.size(), f) != vga_bytes.size())
1687 return -1;
1688 const u32 lo = powerstorm_read_u32(f), hi = powerstorm_read_u32(f),
1689 n = powerstorm_read_u32(f);
1690 if (n < CRealImage2100::MinStateSize || n > CRealImage2100::MaxStateSize)
1691 return -1;
1692 std::string bytes(n, '\0');
1693 if (fread(&bytes[0], 1, n, f) != n)
1694 return -1;
1695 std::istringstream stream(bytes, std::ios::binary);
1696 m_realimage.RestoreState(stream);
1697 if (stream.peek() != std::char_traits<char>::eof())
1698 return -1;
1699 restore_vga_state(vga_bytes);
1700 m_access_count = u64(lo) | (u64(hi) << 32);
1701 // IRQ output is committed after all devices restore.
1702 return 0;
1703 }
1704 catch (const std::exception& e)
1705 {
1706 printf(
1707 "%s: Could not restore graphics state: %s.\n",
1708 devid_string,
1709 e.what());
1710 return -1;
1711 }
