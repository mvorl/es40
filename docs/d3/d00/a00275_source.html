59struct rectangle
60{
61 int min_x = 0, max_x = 0;
62 int min_y = 0, max_y = 0;
63
64 rectangle() = default;
65 rectangle(int x0, int x1, int y0, int y1)
66 : min_x(x0), max_x(x1), min_y(y0), max_y(y1) {
67 }
68
69 int width() const { return max_x - min_x + 1; }
70 int height() const { return max_y - min_y + 1; }
71
72 constexpr bool contains(int32_t x, int32_t y) const { return (x >= min_x) && (x <= max_x) && (y >= min_y) && (y <= max_y); }
73
74 void set(int x0, int x1, int y0, int y1) {
75 min_x = x0; max_x = x1; min_y = y0; max_y = y1;
76 }
77};

79class bitmap_rgb32
80{
81public:
82 bitmap_rgb32() = default;
83
84 void allocate(int w, int h)
85 {
86 m_width = w;
87 m_height = h;
88 m_pixels.resize((size_t)w * h, 0);
89 }
90
91 void resize(int w, int h) { allocate(w, h); }
92
93 int width() const { return m_width; }
94 int height() const { return m_height; }
95 bool valid() const { return m_width > 0 && m_height > 0; }
96
97 // pix(y, x) — returns a reference to pixel (x, y).
98 uint32_t& pix(int y, int x = 0)
99 {
100 return m_pixels[(size_t)y * m_width + x];
101 }
102
103 uint32_t* rowptr(int y)
104 {
105 return m_pixels.data() + (size_t)y * m_width;
106 }
107 const uint32_t* rowptr(int y) const
108 {
109 return m_pixels.data() + (size_t)y * m_width;
110 }
111
112 void fill(uint32_t color, const rectangle& clip)
113 {
114 int y0 = (std::max)(clip.min_y, 0);
115 int y1 = (std::min)(clip.max_y, m_height - 1);
116 int x0 = (std::max)(clip.min_x, 0);
117 int x1 = (std::min)(clip.max_x, m_width - 1);
118 for (int y = y0; y <= y1; y++)
119 {
120 uint32_t* row = &pix(y);
121 for (int x = x0; x <= x1; x++)
122 row[x] = color;
123 }
124 }
125
126 // MAME text cursor: plot a horizontal box
127 void plot_box(int x, int y, int w, int h, uint32_t color)
128 {
129 for (int dy = 0; dy < h; dy++)
130 {
131 if (y + dy < 0 || y + dy >= m_height) continue;
132 uint32_t* row = &pix(y + dy);
133 for (int dx = 0; dx < w; dx++)
134 {
135 if (x + dx >= 0 && x + dx < m_width)
136 row[x + dx] = color;
137 }
138 }
139 }
140
141 // Direct pixel pointer access (whole buffer)
142 const uint32_t* raw() const { return m_pixels.data(); }
143 uint32_t* raw() { return m_pixels.data(); }
144
145 // Clip rect covering the entire bitmap
146 rectangle cliprect() const { return rectangle(0, m_width - 1, 0, m_height - 1); }
147
148private:
149 int m_width = 0;
150 int m_height = 0;
151 std::vector<uint32_t> m_pixels;
152};
